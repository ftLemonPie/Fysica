% VRAAG 1: REKENEN ZONDER GETALLEN
\begin{question}
    Een geladen staaf oefent een kracht uit op een andere geladen staaf. Hoe verandert de kracht wanneer de afstand tussen beide staven wordt verdubbeld?

    \begin{solution}\nl
        \underline{Gegevens:}\\
        Conceptuele vraag. $F_{c}\propto \tfrac{1}{r^{2}}$.\\
        
            \begin{align*}
                &r_{\text{oud}}=r' & & r_{\text{nieuw}}=2\cdot r'\\
                &Q_{1,\text{oud}} =Q_{1,\text{nieuw}} = Q_{1}' & & Q_{2,\text{oud}} =Q_{2,\text{nieuw}} = Q_{2}'
            \end{align*}
        
    \underline{Gevraagd:} verband tussen $F_{c,\text{oud}}$ en $F_{c,\text{nieuw}}$ \\

    \underline{Oplossing:}\\

    We gebruiken de formule van Coulomb:
        \begin{align*}
            F_{c,\text{nieuw}}&=\frac{k\cdot\lvert Q_{1,\text{nieuw}}\rvert\cdot\lvert Q_{2,\text{nieuw}}\rvert}{r_{\text{nieuw}}^{2}}\\
            &=\frac{k\cdot\lvert Q'_{1}\rvert\cdot\lvert Q'_{2}\rvert}{(2\cdot r')^{2}}\\
            &=\frac{k\cdot\lvert Q_{1,\text{oud}}\rvert\cdot\lvert Q_{2,\text{oud}}\rvert}{(2\cdot r_{\text{oud}})^{2}}\\
            &=\frac{1}{2^{2}}\cdot\frac{k\cdot\lvert Q_{1,\text{oud}}\rvert\cdot\lvert Q_{2,\text{oud}}\rvert}{r_{\text{oud}}^{2}}\\
            &=\frac{1}{4}\cdot F_{c,\text{oud}}
        \end{align*}
    De kracht wordt dus vier keer kleiner.\\
    \end{solution}

\end{question}

% VRAAG 2: REKENEN MET EENHEDEN
\begin{question}
    Geef de eenheid van de elektrische kracht volgens de wet van Coulomb en leid deze af uit de formule.

        \begin{solution}\nl
            \underline{Gegevens:}\\
            Wet van Coulomb: $F_{c}=\dfrac{k\cdot Q_{1}Q_{2}}{r^{2}}$

            \underline{Gevraagd:} eenheid $[F_{c}]$ \\

            \underline{Oplossing:}\\
            De constante $k$ heeft als eenheid $\mathrm{\frac{N\cdot m^2}{C^2}}$.  

            \begin{align*}
                [F_{c}] &= \frac{\mathrm{\frac{N\cdot m^2}{C^2}} \cdot \mathrm{C^2}}{\mathrm{m^2}} = \frac{\mathrm{N \cdot m^2}}{\mathrm{m^2}} = \mathrm{N}
            \end{align*}
            De eenheid van de coulombkracht is dus de \textbf{newton (N)}.\\
        \end{solution}
\end{question}

% VRAAG 3: REKENEN ZONDER GETALLEN
\begin{question}
    Wat gebeurt er met de coulombkracht als de afstand tussen twee ladingen driemaal zo groot wordt en beide ladingen verdubbelen in grootte?

    \begin{solution}\nl
        \underline{Gegevens:}\\
        $\begin{array}{@{}w{l}{0.5\textwidth}@{}w{l}{0.5\textwidth}@{}}
            r_{\text{oud}}=r' & r_{\text{nieuw}}=3\cdot r'\\
            Q_{1,\text{oud}} = Q'_{1} & Q_{2,\text{oud}} =Q'_{2}\\
            Q_{1,\text{nieuw}} = 2\cdot Q'_{1}  & Q_{2,\text{nieuw}} = 2\cdot Q'_{2} 
        \end{array}$

        \underline{Gevraagd:} verband tussen $F_{c,\text{oud}}$ en $F_{c,\text{nieuw}}$ \\

        \underline{Oplossing:}\\
        \begin{align*}
            F_{c,\text{nieuw}}&=\frac{k\cdot\lvert Q_{1,\text{nieuw}}\rvert\cdot\lvert Q_{2,\text{nieuw}}\rvert}{r_{\text{nieuw}}^{2}}\\
            &=\frac{k\cdot\lvert 2\cdot Q'_{1}\rvert\cdot\lvert 2\cdot Q'_{2}\rvert}{(3\cdot r')^{2}}\\
            &=\frac{k\cdot\lvert 2\cdot Q_{1,\text{oud}}\rvert\cdot\lvert 2\cdot Q_{2,\text{oud}}\rvert}{(3\cdot r_{\text{oud}})^{2}}\\
            &=\frac{2\cdot 2}{3^{2}}\cdot\frac{k\cdot\lvert Q_{1,\text{oud}}\rvert\cdot\lvert Q_{2,\text{oud}}\rvert}{r_{\text{oud}}^{2}}\\
            &=\frac{4}{9}\cdot F_{c,\text{oud}}
        \end{align*}
        De kracht wordt dus $\frac{4}{9} $ keer zo groot, oftewel 2,25 keer zo klein.\\
    \end{solution}
\end{question}

% VRAAG 4: AANTREKKEN NEUTRAAL VW
\begin{question}
    Leg uit waarom een neutraal voorwerp toch kan worden aangetrokken door een geladen voorwerp.

    \begin{solution}\nl
        Een neutraal voorwerp bevat evenveel positieve als negatieve ladingen.

        Wanneer een geladen voorwerp in de buurt komt, verschuiven de ladingscentra in de moleculen (polarisatie). Hierdoor komt de tegengestelde lading dichter bij het geladen voorwerp te liggen dan de gelijksoortige lading. De aantrekkende kracht is daardoor groter dan de afstotende kracht ($F_{c}\propto \tfrac{1}{r^{2}} $).

        Dit verklaart waarom een neutraal voorwerp toch wordt aangetrokken.\\
    \end{solution}
\end{question}

% VRAAG 5: BALLON
\begin{question}
    Een geladen ballon wordt tegen een muur gehouden en blijft tijdelijk plakken. Hoe kan dit fenomeen verklaard worden?

    \begin{solution}\nl
        De geladen ballon veroorzaakt een polarisatie in de moleculen van de muur (want de muur is meestal een isolator). Daardoor verschuiven de ladingen in de muur, zodat de tegengestelde lading dichter bij de ballon komt te liggen. De resulterende coulombkracht zorgt ervoor dat de ballon aangetrokken wordt en blijft kleven.

        Er geldt $F_{c}\propto \tfrac{1}{r^{2}} $ en ladingen met een tegengesteld teken liggen dichter bij dan ladingen met dezelfde lading, dus de aantrekkende kracht is groter dan de afstotende kracht.

        Als de wrijvingskracht dat ontstaat door de aantrekkende kracht tussen de ballon en de muur groter is dan de zwaartekracht op de ballon, blijft de ballon hangen aan de muur.
    \end{solution}
\end{question}

% VRAAG 6: DAGELIJKS LEVEN
\begin{question}
    Waarom ervaren we in het dagelijks leven de Coulombkrachten tussen voorwerpen meestal niet, hoewel deze in theorie zeer groot kunnen zijn?

    \begin{solution}\nl
        Voorwerpen zijn in het dagelijks leven meestal elektrisch neutraal: ze bevatten evenveel positieve als negatieve ladingen. De Coulombkrachten van de afzonderlijke ladingen heffen elkaar dan op. Alleen bij een overschot of tekort aan elektronen ontstaat er een merkbare kracht.\\
    \end{solution}
\end{question}

% VRAAG 7: GEÏSOLEERDE GELEIDERS
\begin{question}
    4.	Je plaatst drie geïsoleerde geleiders in elkaars nabijheid (van de grond) zoals op figuur~\ref{oef:3geleiders}. Geleider A wordt positief geladen en C wordt geaard. Dan geldt:
    \begin{center}
        \includegraphics[width=5cm]{pic5/elektriciteit/coulomb/oef_3_geleiders.png}
        \label{oef:3geleiders}
    \end{center}
    \begin{multipleChoice}
        \choice{Geleider \textit{B} wordt positief geladen.}
        \choice{Geleider \textit{B} wordt negatief geladen.}
        \choice[correct]{Geleider \textit{B} blijft neutraal.}
        \choice{Geleider \textit{C} wordt positief geladen.}
        \choice[correct]{Geleider \textit{C} wordt negatief geladen.}
        \choice{Geleider \textit{C} blijft neutraal.}        
        \choice{Je kunt geen uitspraak doen over geleiders \textit{B} en \textit{C}.}
    \end{multipleChoice}

    \begin{solution}\nl
        \begin{itemize}
            \item[B] Geleider \textit{B} is geïsoleerd van al de rest en blijft dus neutraal.
            \item[C]  Geleider \textit{A} is positief geladen en zal in geleider \textit{B} de elektronen aantrekken. Hierdoor zal in de linker kant van geleider \textit{B} een negatieve lading zijn en rechts van geleider \textit{B} een positieve lading (een tekort van negatieve ladingen, want geleider \textit{B} blijft neutraal). De positieve kant van geleider \textit{B} trekt zo elektronen aan in geleider \textit{C}, hierdoor zal aan de rechterkant van geleider \textit{C} een tekort aan elektronen ontstaan. Doordat geleider \textit{C} verbonden is met de aarde, zullen er elektronen van de aarde naar geleider \textit{C} gaan, waardoor geleider \textit{C} nu negatief geladen wordt.
        \end{itemize}
    \end{solution}
\end{question}
