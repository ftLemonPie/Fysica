\begin{abstract}
    De potentiële elektrische energie van een lading in een homogeen elektrisch veld, in analogie met de potentiële zwaarte-energie.
\end{abstract}
\maketitle

\section{Referentiepunt en potentiële energie}

\begin{remark}[expandable=true,title={Herhaling: potentiële zwaarte-energie}]\nl
Een massa $m$ die zich op een hoogte $h$ boven het aardoppervlak bevindt, bezit potentiële zwaarte-energie:
\[
    E_{\text{pot}} = m \cdot g \cdot h
\]
Hierbij is het aardoppervlak het gekozen referentiepunt (RP), waar $E_{\text{pot}} = 0 \, \mathrm{J}$. Zie ook appendix~\verwijs[Kinetische en potentiële energie]{app:energievormen}{appendix_energie_vormen}.
\end{remark}

Ook een lading die zich in een elektrisch veld bevindt, bezit energie: \textbf{potentiële elektrische energie}. Van welke factoren hangt die energie af en hoe kan je dat wiskundig weergeven? Om deze energie te kunnen berekenen, moeten we net als bij het zwaarteveld een RP\footnote{RP: referentiepunt.} kiezen.
Een RP is een punt in het elektrisch veld waar we de potentiële energie gelijkstellen aan nul.
Bij een \textbf{homogeen} elektrisch veld, tussen twee evenwijdige platen, kiezen we het RP op de \textbf{negatieve plaat}, zoals te zien in figuur~\ref{fig:rust_potentiaal}.

De potentiële elektrische energie van een lading $Q$ in een punt \textit{P} is de arbeid die de elektrische kracht verricht wanneer de lading van dat punt \textit{P} naar het RP (de negatieve plaat) beweegt.

\section{Afleiding van de formule}
We berekenen die arbeid stap voor stap, voor een lading in een punt $P$ op een afstand $d$ van de negatieve plaat (figuur~\ref{fig:afleiding_epot}).

\begin{itemize}
    \item[\textbf{Stap 1.}] \textbf{De kracht.} In een homogeen veld met veldsterkte $E$ is de grootte van de elektrische kracht op een lading $Q$ gelijk aan $F = \lvert Q \rvert \cdot E$, zoals behandeld in het hoofdstuk \textit{Elektrisch veld}. Deze kracht is overal tussen de platen even groot.
    \item[\textbf{Stap 2.}] \textbf{De verplaatsing.} Van $P$ naar het RP legt de lading een afstand $\Delta x = d$ af, in de zin van de negatieve plaat.
    \item[\textbf{Stap 3.}] \textbf{De arbeid.} Voor een constante kracht geldt $W = +F \cdot \Delta x$ als kracht en verplaatsing dezelfde zin hebben, en $W = -F \cdot \Delta x$ als ze een tegengestelde zin hebben (zie appendix~\verwijs[Arbeid]{app:arbeid}{appendix_energie_arbeid}).
    \begin{itemize}
        \item Een \textbf{positieve} lading wordt naar de negatieve plaat getrokken: kracht en verplaatsing hebben dezelfde zin, dus
        \[
            W = +F \cdot d = +\lvert Q \rvert \cdot E \cdot d = Q \cdot E \cdot d
        \]
        \item Een \textbf{negatieve} lading wordt naar de positieve plaat getrokken: kracht en verplaatsing hebben een tegengestelde zin, dus
        \[
            W = -F \cdot d = -\lvert Q \rvert \cdot E \cdot d = Q \cdot E \cdot d \qquad \text{want } Q = -\lvert Q \rvert
        \]
    \end{itemize}
    \item[\textbf{Stap 4.}] \textbf{De potentiële energie.} In beide gevallen is de arbeid van $P$ naar het RP gelijk aan $Q \cdot E \cdot d$. Dat is de potentiële elektrische energie in $P$.
\end{itemize}

\begin{proposition}[foldable=true,title={Potentiële elektrische energie in een homogeen veld}]\nl
    De potentiële elektrische energie van een lading $Q$ in een punt \textit{P} op een afstand $d$ van de negatieve plaat, in een homogeen elektrisch veld met grootte $E$, wordt gegeven door:
    \begin{equation}\label{eq:epot_homogeen}
        E_{\text{pot}} = Q \cdot E \cdot d
    \end{equation}
    Deze formule geldt voor zowel positieve als negatieve ladingen $Q$ (met behoud van teken). $E_{\text{pot}}$ kan dus zowel positief als negatief zijn.
\end{proposition}

\begin{remark}[expandable=true,title={Eenheid van de potentiële energie}]\nl
De eenheid van $Q \cdot E \cdot d$ is
\[
    \mathrm{C} \cdot \frac{\mathrm{N}}{\mathrm{C}} \cdot \mathrm{m} = \mathrm{N} \cdot \mathrm{m} = \mathrm{J}
\]
want $1 \, \mathrm{N} \cdot \mathrm{m} = 1 \, \mathrm{J}$, de eenheid van arbeid en energie.
\end{remark}

\begin{remark}[expandable=true,title={Notatie}]\nl
Voor de potentiële elektrische energie vind je verschillende notaties: $E_{\text{pot}}$, $E_{p,e}$ of $E_{\text{pot},e}$. De index $e$ wijst op \emph{elektrisch}, om het onderscheid te maken met de potentiële zwaarte-energie. Alle drie de notaties betekenen hetzelfde. In deze cursus gebruiken we $E_{\text{pot}}$.
\end{remark}

\begin{remark}[expandable=true,title={Vergelijking met het zwaarteveld}]\nl
\begin{center}
\begin{tabular}{ccc}
    $m \to Q$ & $g \to E$ & $h \to d$
\end{tabular}
\end{center}
De formule $E_{\text{pot}} = Q \cdot E \cdot d$ is dus volledig analoog aan $E_{\text{pot}} = m \cdot g \cdot h$. Er is één belangrijk verschil: een massa is steeds positief, maar een lading $Q$ kan zowel positief als negatief zijn. Daardoor kan ook $E_{\text{pot}}$ negatief zijn.
\end{remark}

\begin{example}[foldable=true,title={Voorbeeld: teken van de potentiële energie}]\nl
Tussen twee platen op $0{,}15 \, \mathrm{m}$ van elkaar heerst een homogeen veld met grootte $E = 500 \, \mathrm{N/C}$. Bereken de potentiële energie van een lading die zich op de positieve plaat bevindt, voor
\begin{itemize}
    \item[a)] $Q_{1} = +2{,}0 \, \mathrm{\mu C}$
    \item[b)] $Q_{2} = -3{,}0 \, \mathrm{\mu C}$
\end{itemize}

\begin{solution}\nl
\underline{Gegevens:}
\begin{align*}
    & E = 500 \, \mathrm{N/C} & & d = 0{,}15 \, \mathrm{m}
\end{align*}

\underline{Gevraagd:} $E_{\text{pot}}$ voor $Q_1$ en $Q_2$

\underline{Oplossing:}\\
a) \begin{align*}
    E_{\text{pot},1} &= Q_{1} \cdot E \cdot d = 2{,}0 \cdot 10^{-6} \, \mathrm{C} \cdot 500 \, \frac{\mathrm{N}}{\mathrm{C}} \cdot 0{,}15 \, \mathrm{m} \\
    &= \boxed{1{,}5 \cdot 10^{-4} \, \mathrm{J}}
\end{align*}
b) \begin{align*}
    E_{\text{pot},2} &= Q_{2} \cdot E \cdot d = -3{,}0 \cdot 10^{-6} \, \mathrm{C} \cdot 500 \, \frac{\mathrm{N}}{\mathrm{C}} \cdot 0{,}15 \, \mathrm{m} \\
    &= -2{,}25 \cdot 10^{-4} \, \mathrm{J}\simeq \boxed{-2{,}3 \cdot 10^{-4} \, \mathrm{J}}
\end{align*}
\end{solution}
\end{example}

\section{Spontane en geforceerde bewegingen}
Beweegt een lading van een punt $1$ naar een punt $2$, dan is de arbeid van de elektrische kracht het verschil van de arbeid tot het RP vanuit beide punten: $W_{1 \to 2} = E_{\text{pot}}(1) - E_{\text{pot}}(2)$. De verandering van de potentiële energie is dus de tegengestelde van de arbeid:
\[
    \Delta E_{\text{pot}} = E_{\text{pot}}(2) - E_{\text{pot}}(1) = -W
\]
Verricht de elektrische kracht positieve arbeid, dan daalt de potentiële energie. Verricht ze negatieve arbeid, dan stijgt de potentiële energie.

Een beweging is \textbf{spontaan} wanneer de elektrische kracht de lading zelf in die richting drijft. De kracht verricht dan positieve arbeid en de potentiële energie neemt af. Een beweging die niet spontaan kan gebeuren, noemen we \textbf{geforceerd}. Om ze uit te voeren moet een externe kracht arbeid verrichten tegen de elektrische kracht in. Daarom neemt de potentiële energie toe.

\begin{proposition}[foldable=true,title={Spontane beweging in een homogeen veld}]\nl
\begin{itemize}
    \item Een \textbf{positieve} lading beweegt spontaan van de positieve naar de negatieve plaat. In de omgekeerde zin is de beweging geforceerd.
    \item Een \textbf{negatieve} lading beweegt spontaan van de negatieve naar de positieve plaat. In de omgekeerde zin is de beweging geforceerd.
\end{itemize}
Bij een spontane beweging daalt de potentiële energie, bij een geforceerde beweging stijgt ze.
\end{proposition}

Voor een positieve lading die van de positieve naar de negatieve plaat beweegt, wijst de kracht naar de negatieve plaat, net als de verplaatsing. Omdat de afstand $d$ tot het RP afneemt, daalt ook $E_{\text{pot}}$ volgens vergelijking~\ref{eq:epot_homogeen}. Beweegt een negatieve lading van de positieve naar de negatieve plaat, dan wijst de elektrische kracht tegen de verplaatsing in. Er moet dus een externe kracht werken en de potentiële energie stijgt: met $Q < 0$ wordt $E_{\text{pot}} = Q \cdot E \cdot d$ minder negatief als $d$ kleiner wordt.

\begin{question}
    Een positieve lading beweegt in een homogeen elektrisch veld van de positieve naar de negatieve plaat. Welke uitspraken over haar potentiële elektrische energie en over het elektrische veld tussen de platen zijn juist?

    \begin{selectAll}
        \choice{Het elektrische veld wordt sterker.}
        \choice[correct]{Het elektrische veld blijft gelijk.}
        \choice{Het elektrische veld wordt zwakker.}
        \choice{De potentiële energie van de lading stijgt.}
        \choice[correct]{De potentiële energie van de lading daalt.}
        \choice{De potentiële energie van de lading blijft gelijk.}
    \end{selectAll}

    \begin{solution}\nl
        \textbf{Met redeneren:} de positieve lading wordt aangetrokken door de negatieve plaat. De elektrische kracht en de verplaatsing hebben dezelfde zin: de beweging is spontaan en de kracht verricht positieve arbeid. Dus daalt de potentiële energie.

        \textbf{Met de formule:} de lading beweegt naar de negatieve plaat, dus de afstand $d$ tot het RP wordt kleiner. Met $Q>0$ en $E>0$ wordt $E_{\text{pot}} = Q \cdot E \cdot d$ dan kleiner. Dus daalt de potentiële energie.

        In een homogeen veld is de veldsterkte $E$ overal even groot en verandert ze niet wanneer de lading beweegt. Het veld blijft dus gelijk.
    \end{solution}
\end{question}

    \item Is $Q > 0$, dan is de richtingscoëfficiënt positief: de rechte is \textbf{stijgend}.

\end{itemize}


\begin{remark}[expandable=true,title={Richtingscoëfficiënt van de grafiek}]\nl
    $E$ is de grootte van het elektrisch veld en is per definitie altijd positief. Het teken van de richtingscoëfficiënt wordt dus volledig bepaald door het teken van $Q$.
\end{remark}
