\begin{conclusion}[title={Samenvatting}]\nl

    % Potentiële energie in homogeen veld
    \samenbox{Potentiële elektrische energie in een homogeen veld}{
        De potentiële elektrische energie van een lading $Q$ op een afstand $d$ van de negatieve plaat, in een homogeen elektrisch veld met grootte $E$, is:
        \begin{equation*}
            E_{\text{pot}} = Q \cdot E \cdot d
        \end{equation*}
        Het referentiepunt (RP), waar $E_{\text{pot}} = 0 \, \mathrm{J}$, ligt op de negatieve plaat. Voor een negatieve lading is $E_{\text{pot}} < 0$. Bij een spontane beweging daalt de potentiële energie: $\Delta E_{\text{pot}} = -W$.
    }

    % Potentiaal
    \samenbox{Potentiaal}{
        De potentiaal $V$ in een punt is de potentiële energie per eenheid van lading:
        \begin{equation*}
            V = \frac{E_{\text{pot}}}{Q}
        \end{equation*}
        Potentiaal is een scalaire grootheid, uitgedrukt in joule per coulomb ($\mathrm{J/C}$). In een homogeen veld geldt $V = E \cdot d$.
    }

    % Spanning
    \samenbox{Spanning}{
        De spanning tussen twee punten $1$ en $2$ is het verschil in potentiaal:
        \begin{equation*}
            U_{12} = V(1) - V(2)
        \end{equation*}
        Spanning (of potentiaalverschil) wordt uitgedrukt in volt, met $1 \, \mathrm{V} = 1 \, \mathrm{J/C}$. Over een geleider met lengte $\ell$ in een homogeen veld met grootte $E$ geldt $U = E \cdot \ell$.
    }

    % Spanning en energie
    \samenbox{Spanning, arbeid en energie}{
        De spanning $U_{12}$ is de arbeid die het elektrisch veld verricht per eenheid van lading, voor een lading die van $1$ naar $2$ beweegt:
        \begin{equation*}
            U_{12} = \frac{W_{1 \to 2}}{Q} \quad \text{ofwel} \quad W = Q \cdot U_{12}
        \end{equation*}
    }

\end{conclusion}
