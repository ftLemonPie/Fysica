\begin{definition}[foldable=true,title={Spanning}]\nl
De \textbf{spanning} $U_{12}$ tussen twee punten $1$ en $2$ in een elektrisch veld is het verschil in potentiaal tussen die twee punten:
\begin{equation}\label{eq:spanning_definitie}
    U_{12} = V(1) - V(2)
\end{equation}
Spanning wordt uitgedrukt in \textbf{volt} (V), genoemd naar Alessandro Volta (1745-1827). Er geldt $1 \, \mathrm{V} = 1 \, \mathrm{J/C}$.
\end{definition}

Spanning noemen we ook \textbf{potentiaalverschil}: beide woorden betekenen hetzelfde.

\begin{remark}[expandable=true,title={Potentiaal in J/C, spanning in V}]\nl
Hoewel $1 \, \mathrm{V} = 1 \, \mathrm{J/C}$, drukken we in deze cursus de potentiaal uit in $\mathrm{J/C}$ en de spanning in $\mathrm{V}$. Zo zie je meteen of het over de potentiaal in \emph{één} punt gaat, of over de spanning \emph{tussen twee} punten. Bovendien vermijd je verwarring tussen het symbool $V$ van de potentiaal en de eenheid $\mathrm{V}$ (volt).
\end{remark}

\begin{remark}[expandable=true,title={Spanning is een verschil}]\nl
Spanning is een verschil en kan dus zowel positief als negatief zijn, afhankelijk van welke twee punten je met elkaar vergelijkt en in welke volgorde.
\end{remark}

\begin{question}
    Punt $1$ heeft een potentiaal van $12 \, \mathrm{J/C}$ en punt $2$ een potentiaal van $5 \, \mathrm{J/C}$. Hoeveel bedraagt de spanning $U_{12}$?

    \begin{multipleChoice}
        \choice[correct]{$+7 \, \mathrm{V}$}
        \choice{$-7 \, \mathrm{V}$}
        \choice{$+17 \, \mathrm{V}$}
        \choice{$+2{,}4 \, \mathrm{V}$}
    \end{multipleChoice}

    \begin{solution}\nl
        $U_{12} = V(1) - V(2) = 12 \, \mathrm{J/C} - 5 \, \mathrm{J/C} = 7 \, \mathrm{J/C} = +7 \, \mathrm{V}$. Spanning is een verschil van potentialen, geen som en geen quotiënt.
    \end{solution}
\end{question}

\section{De polen van een spanningsbron}
Een \textbf{spanningsbron}, bijvoorbeeld een batterij, onderhoudt een spanning tussen haar twee polen. De \textbf{pluspool} ($+$) is de pool met de hoogste potentiaal, de \textbf{minpool} ($-$) de pool met de laagste potentiaal. Meer over het meten van spanning en over de soorten spanningsbronnen vind je in appendix~\verwijs[Meten van spanning en spanningsbronnen]{hoofdstuk:appendix_spanning}{appendix_spanning_inleiding}.

\begin{question}
    Twee identieke lampjes staan in serie met een batterij van $12 \, \mathrm{V}$. Wat is de spanning over één lampje?

    \begin{multipleChoice}
        \choice{$12 \, \mathrm{V}$}
        \choice[correct]{$6 \, \mathrm{V}$}
        \choice{$3 \, \mathrm{V}$}
        \choice{$0 \, \mathrm{V}$}
    \end{multipleChoice}

    \begin{solution}\nl
        De spanningen over de twee lampjes samen zijn gelijk aan de spanning van de batterij. Omdat de lampjes identiek zijn, is de spanning over elk lampje de helft: $6 \, \mathrm{V}$.
    \end{solution}
\end{question}

\section{Spanning over een geleider}
Beschouw opnieuw het homogeen veld tussen twee platen, met veldsterkte $E$. Een punt $P$ op een afstand $x$ van de negatieve plaat heeft potentiaal $V(P) = E \cdot x$.

\begin{proposition}[foldable=true,title={Spanning over een geleider}]\nl
Voor de spanning $U$ over een geleider met lengte $\ell$, waarin een homogeen elektrisch veld met grootte $E$ heerst, geldt:
\begin{equation}\label{eq:spanning_geleider}
    U = E \cdot \ell
\end{equation}
\end{proposition}

\begin{example}[foldable=true,title={Voorbeeld: spanning over een geleider}]\nl
In een geleider met lengte $0{,}20 \, \mathrm{m}$ heerst een homogeen elektrisch veld met grootte $250 \, \mathrm{N/C}$. Bereken de spanning over de geleider.

\begin{solution}\nl
\underline{Gegevens:}
\begin{align*}
    & E = 250 \, \mathrm{N/C} & & \ell = 0{,}20 \, \mathrm{m}
\end{align*}

\underline{Gevraagd:} $U$

\underline{Oplossing:}
\begin{align*}
    U = E \cdot \ell = 250 \, \frac{\mathrm{N}}{\mathrm{C}} \cdot 0{,}20 \, \mathrm{m} = 50 \, \mathrm{V} = \boxed{5{,}0 \cdot 10^{1} \, \mathrm{V}}
\end{align*}
\end{solution}
\end{example}

\begin{remark}[expandable=true,title={Veldsterkte uit spanning}]\nl
Uit $U = E \cdot \ell$ volgt ook $E = U/\ell$: hoe groter de opgelegde spanning over een geleider met vaste lengte, hoe groter het elektrisch veld erin. Daarom is de eenheid $\mathrm{N/C}$ van het elektrisch veld hetzelfde als $\mathrm{V/m}$. Beide eenheden worden in de praktijk gebruikt.
\end{remark}

\begin{question}
    Een spanningsbron van $12 \, \mathrm{V}$ blijft aangesloten op twee evenwijdige platen. Je schuift de platen dichter naar elkaar toe, zodat hun onderlinge afstand halveert. Wat gebeurt er met de grootte van het elektrisch veld tussen de platen?

    \begin{multipleChoice}
        \choice{Ze halveert.}
        \choice{Ze blijft gelijk.}
        \choice[correct]{Ze verdubbelt.}
    \end{multipleChoice}

    \begin{solution}\nl
        De spanning $U$ blijft $12 \, \mathrm{V}$ omdat de bron aangesloten blijft. Uit $E = U/\ell$ volgt dat het veld verdubbelt wanneer de afstand $\ell$ halveert.
    \end{solution}
\end{question}
