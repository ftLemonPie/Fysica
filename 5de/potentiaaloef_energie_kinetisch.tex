%———————————————————————————————————————
% OEF 1: proton versneld vanuit rust
\begin{exercise}
    Een proton, aanvankelijk in rust, wordt versneld door een spanning van $400 \, \mathrm{V}$. Bereken de snelheid van het proton nadat het deze spanning heeft doorlopen.

    \[ v = \answer[tolerance=7000,onlineshowanswerbutton]{277000} \, \mathrm{m/s} \]

    \begin{hint}
        Waarin wordt de arbeid van het elektrisch veld omgezet? De massa en de lading van een proton vind je in je tabellenboekje.
    \end{hint}

    \begin{solution}\nl
        \underline{Gegevens:}
        \begin{align*}
            & U = 400 \, \mathrm{V} & & Q = e = 1{,}60 \cdot 10^{-19} \, \mathrm{C} \\
            & m_{p} = 1{,}67 \cdot 10^{-27} \, \mathrm{kg} & & v_{\text{begin}} = 0 \, \mathrm{m/s}
        \end{align*}

        \underline{Gevraagd:} $v$

        \underline{Oplossing:}
        \begin{align*}
            Q \cdot U &= \tfrac{1}{2} m_{p} v^{2} \\
            1{,}60 \cdot 10^{-19} \, \mathrm{C} \cdot 400 \, \mathrm{V} &= \tfrac{1}{2} \cdot 1{,}67 \cdot 10^{-27} \, \mathrm{kg} \cdot v^{2} \\
            v &= \sqrt{\frac{2 \cdot 6{,}40 \cdot 10^{-17} \, \mathrm{J}}{1{,}67 \cdot 10^{-27} \, \mathrm{kg}}} \\
            &= \boxed{2{,}77 \cdot 10^{5} \, \mathrm{m/s}}
        \end{align*}
    \end{solution}
\end{exercise}

% OEF 2: elektron met beginsnelheid, versneld door spanning
\begin{exercise}
    Een elektron beweegt met een beginsnelheid van $2{,}0 \cdot 10^{6} \, \mathrm{m/s}$ en doorloopt daarna een spanning $U_{12} = -50 \, \mathrm{V}$ (waarbij het elektron van punt 1 naar punt 2 beweegt). Bereken de eindsnelheid van het elektron.

    \begin{hint}
        Let op het teken van de lading van het elektron. Is de arbeid van het veld positief of negatief?
    \end{hint}
    \begin{hint}
        Het elektron heeft al kinetische energie in het begin. Bereken die eerst.
    \end{hint}

    \[ v_{\text{eind}} = \answer[tolerance=120000,onlineshowanswerbutton]{4600000} \, \mathrm{m/s} \]

    \begin{solution}\nl
        \underline{Gegevens:}
        \begin{align*}
            & v_{\text{begin}} = 2{,}0 \cdot 10^{6} \, \mathrm{m/s} & & U_{12} = -50 \, \mathrm{V} \\
            & Q = -e = -1{,}60 \cdot 10^{-19} \, \mathrm{C} & & m_{e} = 9{,}11 \cdot 10^{-31} \, \mathrm{kg}
        \end{align*}

        \underline{Gevraagd:} $v_{\text{eind}}$

        \underline{Oplossing:}\\
        De beginkinetische energie van het elektron:
        \begin{align*}
            E_{\text{kin,begin}} &= \tfrac{1}{2} m_{e} v_{\text{begin}}^{2} \\
            &= \tfrac{1}{2} \cdot 9{,}11 \cdot 10^{-31} \, \mathrm{kg} \cdot (2{,}0 \cdot 10^{6} \, \mathrm{m/s})^{2} \\
            &= 1{,}822 \cdot 10^{-18} \, \mathrm{J}
        \end{align*}
        De arbeid die het veld op het elektron verricht:
        \begin{align*}
            W &= Q \cdot U_{12} = (-1{,}60 \cdot 10^{-19} \, \mathrm{C}) \cdot (-50 \, \mathrm{V}) \\
            &= 8{,}00 \cdot 10^{-18} \, \mathrm{J}
        \end{align*}
        Uit de arbeid-energiestelling volgt de eindkinetische energie:
        \begin{align*}
            E_{\text{kin,eind}} &= E_{\text{kin,begin}} + W \\
            &= 1{,}822 \cdot 10^{-18} \, \mathrm{J} + 8{,}00 \cdot 10^{-18} \, \mathrm{J} \\
            &= 9{,}822 \cdot 10^{-18} \, \mathrm{J}
        \end{align*}
        Hieruit volgt de eindsnelheid:
        \begin{align*}
            v_{\text{eind}} = \sqrt{\frac{2 \cdot E_{\text{kin,eind}}}{m_{e}}} = \sqrt{\frac{2 \cdot 9{,}822 \cdot 10^{-18} \, \mathrm{J}}{9{,}11 \cdot 10^{-31} \, \mathrm{kg}}} = 4{,}64 \cdot 10^{6} \, \mathrm{m/s} \simeq \boxed{4{,}6 \cdot 10^{6} \, \mathrm{m/s}}
        \end{align*}
        De beginsnelheid en de spanning hebben 2 beduidende cijfers, dus ook het antwoord. De positieve arbeid geeft aan dat het elektron versnelt: de eindsnelheid is dus groter dan de beginsnelheid.
    \end{solution}
\end{exercise}

% OEF 3 (moeilijkst): Millikan-stijl, meerdelige oefening
\begin{exercise}
    Een oliedruppeltje met massa $1{,}96 \cdot 10^{-15} \, \mathrm{kg}$ bevindt zich tussen twee horizontale platen op $5{,}00 \cdot 10^{-3} \, \mathrm{m}$ van elkaar. Het blijft juist zweven bij een spanning van $150 \, \mathrm{V}$ tussen de platen.

    \begin{question}
        Bereken de grootte van het elektrisch veld tussen de platen.

        \[ E = \answer[tolerance=750,onlineshowanswerbutton]{30000} \, \mathrm{N/C} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & U = 150 \, \mathrm{V} & & d = 5{,}00 \cdot 10^{-3} \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $E$

            \underline{Oplossing:}
            \begin{align*}
                E = \frac{U}{d} = \frac{150 \, \mathrm{V}}{5{,}00 \cdot 10^{-3} \, \mathrm{m}} = \boxed{3{,}00 \cdot 10^{4} \, \mathrm{N/C}}
            \end{align*}
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de grootte van de lading op het druppeltje.

        \[ \lvert Q \rvert = \answer[tolerance=0.16,onlineshowanswerbutton]{6.41} \cdot 10^{-19} \, \mathrm{C} \]

        \begin{hint}
            Het druppeltje zweeft, dus de resulterende kracht op het druppeltje is nul: de zwaartekracht en de elektrische kracht zijn even groot.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & m = 1{,}96 \cdot 10^{-15} \, \mathrm{kg} & & g = 9{,}81 \, \mathrm{N/kg} & & E = 3{,}00 \cdot 10^{4} \, \mathrm{N/C}
            \end{align*}

            \underline{Gevraagd:} $\lvert Q \rvert$

            \underline{Oplossing:}\\
            De grootte van de zwaartekracht:
            \begin{align*}
                F_{z} = m \cdot g = 1{,}96 \cdot 10^{-15} \, \mathrm{kg} \cdot 9{,}81 \, \mathrm{N/kg} = 1{,}923 \cdot 10^{-14} \, \mathrm{N}
            \end{align*}
            Omdat het druppeltje zweeft, geldt $F_{z} = F_{e} = E \cdot \lvert Q \rvert$:
            \begin{align*}
                \lvert Q \rvert = \frac{F_{z}}{E} = \frac{1{,}923 \cdot 10^{-14} \, \mathrm{N}}{3{,}00 \cdot 10^{4} \, \mathrm{N/C}} = \boxed{6{,}41 \cdot 10^{-19} \, \mathrm{C}}
            \end{align*}
        \end{solution}
    \end{question}

    \begin{question}
        Hoeveel elektronen heeft het druppeltje opgenomen?

        \[ n = \answer[tolerance=0.1,onlineshowanswerbutton]{4} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & \lvert Q \rvert = 6{,}41 \cdot 10^{-19} \, \mathrm{C} & & e = 1{,}60 \cdot 10^{-19} \, \mathrm{C}
            \end{align*}

            \underline{Gevraagd:} $n$

            \underline{Oplossing:}
            \begin{align*}
                n = \frac{\lvert Q \rvert}{e} = \frac{6{,}41 \cdot 10^{-19} \, \mathrm{C}}{1{,}60 \cdot 10^{-19} \, \mathrm{C}} = \boxed{4}
            \end{align*}
            Het druppeltje heeft dus $4$ elektronen opgenomen. Dit bevestigt dat de lading van het druppeltje een geheel veelvoud is van de elementaire lading, zoals Millikan experimenteel vaststelde.
        \end{solution}
    \end{question}
\end{exercise}
