% BASIS 1: STROOMSTERKTE
\begin{exercise}
    Door een verbindingsdraad stroomt in $2{,}0\,\mathrm{min}$ een lading van $36\,\mathrm{C}$. Bereken de stroomsterkte.

    \[ I = \answer[tolerance=0.05,onlineshowanswerbutton]{0.30} \, \mathrm{A} \]

    \begin{solution}\nl
        \underline{Gegevens:}
        \begin{align*}
            & \Delta Q = 36\,\mathrm{C} & & \Delta t = 2{,}0\,\mathrm{min} = 120\,\mathrm{s}
        \end{align*}

        \underline{Gevraagd:} $I$ \\

        \underline{Oplossing:}
        \begin{align*}
            I &= \frac{\Delta Q}{\Delta t} = \frac{36\,\mathrm{C}}{120\,\mathrm{s}} = \boxed{3{,}0 \cdot 10^{-1}\,\mathrm{A}}
        \end{align*}
    \end{solution}
\end{exercise}
%———————————————————————————————————————

%———————————————————————————————————————
% BASIS 2: OHM, STROOM
\begin{exercise}
    Een weerstand van $0{,}45\,\mathrm{k\Omega}$ wordt aangesloten op een spanning van $9{,}0\,\mathrm{V}$. Bereken de stroomsterkte in $\mathrm{mA}$.

    \[ I = \answer[tolerance=0.05,onlineshowanswerbutton]{20} \, \mathrm{mA} \]

    \begin{solution}\nl
        \underline{Gegevens:}
        \begin{align*}
            & R = 0{,}45\,\mathrm{k\Omega} = 450\,\mathrm{\Omega} & & U = 9{,}0\,\mathrm{V}
        \end{align*}

        \underline{Gevraagd:} $I$ \\

        \underline{Oplossing:}
        \begin{align*}
            I &= \frac{U}{R} = \frac{9{,}0\,\mathrm{V}}{450\,\mathrm{\Omega}} = 0{,}020\,\mathrm{A} = 20\,\mathrm{mA} = \boxed{2{,}0 \cdot 10^{-2}\,\mathrm{A}}
        \end{align*}
    \end{solution}
\end{exercise}
%———————————————————————————————————————

%———————————————————————————————————————
% BASIS 3: OHM, SPANNING
\begin{exercise}
    Door een weerstand van $48\,\mathrm{\Omega}$ loopt een stroom van $250\,\mathrm{mA}$. Bereken de spanning over de weerstand.

    \[ U = \answer[tolerance=0.05,onlineshowanswerbutton]{12} \, \mathrm{V} \]

    \begin{solution}\nl
        \underline{Gegevens:}
        \begin{align*}
            & R = 48\,\mathrm{\Omega} & & I = 250\,\mathrm{mA} = 0{,}250\,\mathrm{A}
        \end{align*}

        \underline{Gevraagd:} $U$ \\

        \underline{Oplossing:}
        \begin{align*}
            U &= R \cdot I = 48\,\mathrm{\Omega} \cdot 0{,}250\,\mathrm{A} = \boxed{1{,}2 \cdot 10^{1}\,\mathrm{V}}
        \end{align*}
    \end{solution}
\end{exercise}
%———————————————————————————————————————

%———————————————————————————————————————
% BASIS 4: OHM, WEERSTAND
\begin{exercise}
    Het verwarmingselement van een strijkijzer is aangesloten op $230\,\mathrm{V}$ en neemt een stroom van $9{,}2\,\mathrm{A}$ op. Bereken de weerstand van het element.

    \[ R = \answer[tolerance=0.6,onlineshowanswerbutton]{25} \, \mathrm{\Omega} \]

    \begin{solution}\nl
        \underline{Gegevens:}
        \begin{align*}
            & U = 230\,\mathrm{V} & & I = 9{,}2\,\mathrm{A}
        \end{align*}

        \underline{Gevraagd:} $R$ \\

        \underline{Oplossing:}
        \begin{align*}
            R &= \frac{U}{I} = \frac{230\,\mathrm{V}}{9{,}2\,\mathrm{A}} = \boxed{2{,}5 \cdot 10^{1}\,\mathrm{\Omega}}
        \end{align*}
    \end{solution}
\end{exercise}
