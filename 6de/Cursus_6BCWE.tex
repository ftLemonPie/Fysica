\section{Verplaatsing, afgelegde weg en gemiddelde snelheid}

%———————————————————————————————————————
% OEFENING 1 (makkelijkst): verplaatsing, afgelegde weg en gemiddelde snelheid
\begin{exercise}
    Een koerier vertrekt in het depot ($x = 0{,}0 \, \mathrm{km}$) en rijdt over een rechte weg langs de $x$-as. Eerst rijdt hij $24 \, \mathrm{km}$ in de positieve zin met $48 \, \mathrm{km/h}$. Daarna wacht hij $15 \, \mathrm{min}$ bij een klant. Tot slot rijdt hij $6{,}0 \, \mathrm{km}$ terug in de negatieve zin met $36 \, \mathrm{km/h}$.

    \begin{question}
        Bereken de verplaatsing $\Delta x$ en de afgelegde weg $s$ over de volledige rit.

        \[ \Delta x = \answer[tolerance=0.01,onlineshowanswerbutton]{18} \, \mathrm{km} \qquad \text{en} \qquad s = \answer[tolerance=0.01,onlineshowanswerbutton]{30} \, \mathrm{km} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x_{\text{begin}} = 0{,}0 \, \mathrm{km} & & \text{heen: } 24 \, \mathrm{km} \text{ in de positieve zin}\\
                & \text{terug: } 6{,}0 \, \mathrm{km} \text{ in de negatieve zin} & &
            \end{align*}

            \underline{Gevraagd:} $\Delta x$ en $s$\\

            \underline{Oplossing:}\\
            De eindpositie is $x_{\text{eind}} = 24 \, \mathrm{km} - 6{,}0 \, \mathrm{km} = 18 \, \mathrm{km}$. De verplaatsing is het verschil tussen eind- en beginpositie:
            \[
                \Delta x = x_{\text{eind}} - x_{\text{begin}} = 18 \, \mathrm{km} - 0{,}0 \, \mathrm{km} = 18 \, \mathrm{km}
            \]
            De afgelegde weg is de volledige afgelegde afstand, ongeacht de zin:
            \[
                s = 24 \, \mathrm{km} + 6{,}0 \, \mathrm{km} = 30 \, \mathrm{km}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde snelheid $\langle v_x \rangle$ over de volledige rit, inclusief de wachttijd, in $\mathrm{m/s}$ en in $\mathrm{km/h}$.

        \[ \langle v_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{5.45} \, \mathrm{m/s} = \answer[tolerance=0.01,onlineshowanswerbutton]{19.6} \, \mathrm{km/h} \]

        \begin{hint}
            Je hebt de totale duur van de rit nodig. Welke drie tijdsintervallen moet je optellen?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & \Delta x = 18 \, \mathrm{km} = 18 \cdot 10^{3} \, \mathrm{m} & & v_{\text{heen}} = 48 \, \mathrm{km/h} = 13{,}333 \, \mathrm{m/s}\\
                & s_{\text{heen}} = 24 \cdot 10^{3} \, \mathrm{m} & & v_{\text{terug}} = 36 \, \mathrm{km/h} = 10 \, \mathrm{m/s}\\
                & s_{\text{terug}} = 6{,}0 \cdot 10^{3} \, \mathrm{m} & & t_{\text{wacht}} = 15 \, \mathrm{min} = 900 \, \mathrm{s}
            \end{align*}

            \underline{Gevraagd:} $\langle v_x \rangle$\\

            \underline{Oplossing:}\\
            We berekenen de duur van elk deel van de rit:
            \begin{align*}
                \Delta t_{\text{heen}} &= \frac{24 \cdot 10^{3} \, \mathrm{m}}{13{,}333 \, \mathrm{m/s}} = 1800 \, \mathrm{s}\\
                \Delta t_{\text{terug}} &= \frac{6{,}0 \cdot 10^{3} \, \mathrm{m}}{10 \, \mathrm{m/s}} = 600 \, \mathrm{s}
            \end{align*}
            De totale duur is $\Delta t = 1800 \, \mathrm{s} + 900 \, \mathrm{s} + 600 \, \mathrm{s} = 3300 \, \mathrm{s}$. De gemiddelde snelheid is de verplaatsing gedeeld door de duur:
            \begin{align*}
                \langle v_x \rangle &= \frac{\Delta x}{\Delta t} = \frac{18 \cdot 10^{3} \, \mathrm{m}}{3300 \, \mathrm{s}}\\
                &= 5{,}45 \, \mathrm{m/s} = 5{,}45 \cdot 3{,}6 \, \mathrm{km/h} = 19{,}6 \, \mathrm{km/h}
            \end{align*}
            Merk op dat dit niet het gemiddelde is van $48 \, \mathrm{km/h}$ en $36 \, \mathrm{km/h}$: de wachttijd en de zin van de beweging tellen mee.
        \end{solution}
    \end{question}

    \begin{question}
        Een boordcomputer meet de afgelegde weg gedeeld door de tijd. Bereken die gemiddelde snelheidsgrootte over de volledige rit.

        \[ \frac{s}{\Delta t} = \answer[tolerance=0.01,onlineshowanswerbutton]{32.7} \, \mathrm{km/h} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & s = 30 \cdot 10^{3} \, \mathrm{m} & & \Delta t = 3300 \, \mathrm{s}
            \end{align*}

            \underline{Gevraagd:} $\dfrac{s}{\Delta t}$\\

            \underline{Oplossing:}\\
            \begin{align*}
                \frac{s}{\Delta t} &= \frac{30 \cdot 10^{3} \, \mathrm{m}}{3300 \, \mathrm{s}} = 9{,}09 \, \mathrm{m/s}\\
                &= 9{,}09 \cdot 3{,}6 \, \mathrm{km/h} = 32{,}7 \, \mathrm{km/h}
            \end{align*}
            Deze waarde is groter dan $\langle v_x \rangle = 19{,}6 \, \mathrm{km/h}$, omdat de afgelegde weg ($30 \, \mathrm{km}$) groter is dan de verplaatsing ($18 \, \mathrm{km}$).
        \end{solution}
    \end{question}

    \begin{question}
        De koerier wil dat zijn gemiddelde snelheid over de volledige rit (inclusief wachttijd) $20 \, \mathrm{km/h}$ bedraagt. Met welke constante snelheid had hij dan terug moeten rijden? De heenrit en de wachttijd blijven ongewijzigd.

        \[ v_{\text{terug}} = \answer[tolerance=0.01,onlineshowanswerbutton]{40} \, \mathrm{km/h} \]

        \begin{hint}
            Hoe lang mag de volledige rit hoogstens duren om op $20 \, \mathrm{km/h}$ uit te komen? Hoeveel van die tijd is al opgebruikt voor het heen en het wachten?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & \Delta x = 18 \cdot 10^{3} \, \mathrm{m} & & \langle v_x \rangle = 20 \, \mathrm{km/h} = 5{,}556 \, \mathrm{m/s}\\
                & \Delta t_{\text{heen}} = 1800 \, \mathrm{s} & & t_{\text{wacht}} = 900 \, \mathrm{s}\\
                & s_{\text{terug}} = 6{,}0 \cdot 10^{3} \, \mathrm{m} & &
            \end{align*}

            \underline{Gevraagd:} $v_{\text{terug}}$\\

            \underline{Oplossing:}\\
            De verplaatsing blijft $18 \cdot 10^{3} \, \mathrm{m}$. De totale duur die bij $\langle v_x \rangle = 5{,}556 \, \mathrm{m/s}$ hoort, is:
            \[
                \Delta t = \frac{\Delta x}{\langle v_x \rangle} = \frac{18 \cdot 10^{3} \, \mathrm{m}}{5{,}556 \, \mathrm{m/s}} = 3240 \, \mathrm{s}
            \]
            Daarvan is al $1800 \, \mathrm{s} + 900 \, \mathrm{s} = 2700 \, \mathrm{s}$ opgebruikt. Voor de terugrit blijft dus $3240 \, \mathrm{s} - 2700 \, \mathrm{s} = 540 \, \mathrm{s}$ over:
            \[
                v_{\text{terug}} = \frac{6{,}0 \cdot 10^{3} \, \mathrm{m}}{540 \, \mathrm{s}} = 11{,}1 \, \mathrm{m/s} = 40 \, \mathrm{km/h}
            \]
        \end{solution}
    \end{question}
\end{exercise}

\section{Grafieken: van de $x(t)$-grafiek naar snelheid en versnelling}
Hier lees je de snelheid af als de helling van een $x(t)$-grafiek of van een raaklijn, en bepaal je daaruit (gemiddelde) versnellingen.

%———————————————————————————————————————
% OEFENING: x(t)-grafiek naar v(t) en a(t)
\begin{exercise}
    Een robotwagentje rijdt over een rechte rail langs de $x$-as. De figuur toont zijn positie $x$ als functie van de tijd.

    \begin{center}
        \begin{tikzpicture}[x=0.45cm,y=0.2cm]
            \draw[gray!40, very thin] (0,-8) grid[xstep=2,ystep=4] (16,12);
            \draw[positie] (0,0) -- (17,0) node[right] {$t\,(\mathrm{s})$};
            \draw[positie] (0,-9.5) -- (0,14.5) node[above] {$x\,(\mathrm{m})$};
            \foreach \tx in {2,4,...,16} \draw (\tx,0.4) -- (\tx,-0.4) node[above=3pt, font=\small] {\tx};
            \foreach \xy in {-8,-4,4,8,12} \draw (0.2,\xy) -- (-0.2,\xy) node[left, font=\small] {$\xy$};
            \draw[line width=1pt, black] (0,0) -- (4,12) -- (8,12) -- (12,0) -- (16,-8);
        \end{tikzpicture}
    \end{center}

    \begin{question}
        Bereken de snelheid $v_x$ van het wagentje tussen $t = 0 \, \mathrm{s}$ en $t = 4{,}0 \, \mathrm{s}$.

        \[ v_x = \answer[tolerance=0.01,onlineshowanswerbutton]{3.0} \, \mathrm{m/s} \]

        \begin{hint}
            Wat stelt de helling van een $x(t)$-grafiek voor?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(0) = 0 \, \mathrm{m} & & x(4{,}0 \, \mathrm{s}) = 12 \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $v_x$\\

            \underline{Oplossing:}\\
            De grafiek is een rechte, dus de snelheid is constant en gelijk aan de helling:
            \[
                v_x = \frac{\Delta x}{\Delta t} = \frac{12 \, \mathrm{m} - 0 \, \mathrm{m}}{4{,}0 \, \mathrm{s} - 0 \, \mathrm{s}} = 3{,}0 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de snelheid $v_x$ tussen $t = 8{,}0 \, \mathrm{s}$ en $t = 12 \, \mathrm{s}$.

        \[ v_x = \answer[tolerance=0.01,onlineshowanswerbutton]{-3.0} \, \mathrm{m/s} \]

        Tussen $t = 4{,}0 \, \mathrm{s}$ en $t = 8{,}0 \, \mathrm{s}$ geldt: \wordChoice{
            \choice[correct]{het wagentje staat stil}
            \choice{het wagentje rijdt met constante snelheid vooruit}
            \choice{het wagentje versnelt}
        }

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(8{,}0 \, \mathrm{s}) = 12 \, \mathrm{m} & & x(12 \, \mathrm{s}) = 0 \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $v_x$\\

            \underline{Oplossing:}\\
            \[
                v_x = \frac{\Delta x}{\Delta t} = \frac{0 \, \mathrm{m} - 12 \, \mathrm{m}}{12 \, \mathrm{s} - 8{,}0 \, \mathrm{s}} = -3{,}0 \, \mathrm{m/s}
            \]
            Het wagentje rijdt hier in de negatieve zin terug naar de oorsprong. Tussen $4{,}0 \, \mathrm{s}$ en $8{,}0 \, \mathrm{s}$ is de grafiek horizontaal: de positie verandert niet, dus $v_x = 0 \, \mathrm{m/s}$ en het wagentje staat stil.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde versnelling $\langle a_x \rangle$ tussen $t = 2{,}0 \, \mathrm{s}$ en $t = 10 \, \mathrm{s}$.

        \[ \langle a_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{-0.75} \, \mathrm{m/s^2} \]

        \begin{hint}
            Je kent de snelheid op beide tijdstippen niet rechtstreeks, maar wel via de grafiek. Op welke momenten moet je dus de snelheid bepalen?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_x(2{,}0 \, \mathrm{s}) = 3{,}0 \, \mathrm{m/s} & & v_x(10 \, \mathrm{s}) = -3{,}0 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} $\langle a_x \rangle$\\

            \underline{Oplossing:}\\
            Op $t = 2{,}0 \, \mathrm{s}$ is het wagentje in het eerste stuk (helling $3{,}0 \, \mathrm{m/s}$), op $t = 10 \, \mathrm{s}$ in het stuk met helling $-3{,}0 \, \mathrm{m/s}$:
            \[
                \langle a_x \rangle = \frac{\Delta v_x}{\Delta t} = \frac{-3{,}0 \, \mathrm{m/s} - 3{,}0 \, \mathrm{m/s}}{10 \, \mathrm{s} - 2{,}0 \, \mathrm{s}} = -0{,}75 \, \mathrm{m/s^2}
            \]
            In werkelijkheid verloopt een overgang tussen twee snelheden nooit ogenblikkelijk. De knikken in de grafiek zijn dus een vereenvoudiging, en daarom rekenen we hier met een gemiddelde versnelling.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde snelheid $\langle v_x \rangle$ tussen $t = 0 \, \mathrm{s}$ en $t = 16 \, \mathrm{s}$.

        \[ \langle v_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{-0.50} \, \mathrm{m/s} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(0) = 0 \, \mathrm{m} & & x(16 \, \mathrm{s}) = -8{,}0 \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $\langle v_x \rangle$\\

            \underline{Oplossing:}\\
            \[
                \langle v_x \rangle = \frac{\Delta x}{\Delta t} = \frac{-8{,}0 \, \mathrm{m} - 0 \, \mathrm{m}}{16 \, \mathrm{s} - 0 \, \mathrm{s}} = -0{,}50 \, \mathrm{m/s}
            \]
            Het wagentje is uiteindelijk $8{,}0 \, \mathrm{m}$ links van de oorsprong, dus de gemiddelde snelheid is negatief.
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING: gekromde x(t)-grafiek, raaklijn
\begin{exercise}
    De positie van een voorwerp op de $x$-as wordt gegeven door
    \[
        x(t) = -1{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t^{2} + 8{,}0 \, \mathrm{\frac{m}{s}} \cdot t
    \]
    met $t$ in seconden. De grafiek is hieronder getekend, samen met de raaklijn aan de grafiek op $t = 2{,}0 \, \mathrm{s}$. Die raaklijn gaat door de punten $(0 \, \mathrm{s}; 4{,}0 \, \mathrm{m})$ en $(3{,}0 \, \mathrm{s}; 16 \, \mathrm{m})$.

    \begin{center}
        \begin{tikzpicture}[x=0.8cm,y=0.25cm]
            \draw[gray!40, very thin] (0,0) grid[xstep=1,ystep=4] (8,17);
            \draw[positie] (0,0) -- (8.8,0) node[right] {$t\,(\mathrm{s})$};
            \draw[positie] (0,0) -- (0,19) node[above] {$x\,(\mathrm{m})$};
            \foreach \tx in {1,2,...,8} \draw (\tx,0.4) -- (\tx,-0.4) node[below=3pt, font=\small] {\tx};
            \foreach \xy in {4,8,12,16} \draw (0.1,\xy) -- (-0.1,\xy) node[left, font=\small] {$\xy$};
            \draw[line width=1pt, black, domain=0:8, samples=40, smooth, variable=\xx] plot ({\xx},{-\xx*\xx+8*\xx});
            \draw[line width=0.8pt, red!100!black] (0,4) -- (3,16);
            \fill[red!100!black] (0,4) circle (2pt) (3,16) circle (2pt);
            \fill (2,12) circle (2pt);
        \end{tikzpicture}
    \end{center}

    \begin{question}
        Bepaal de snelheid $v_x$ op $t = 2{,}0 \, \mathrm{s}$ met behulp van de raaklijn.

        \[ v_x(2{,}0 \, \mathrm{s}) = \answer[tolerance=0.01,onlineshowanswerbutton]{4.0} \, \mathrm{m/s} \]

        \begin{hint}
            De ogenblikkelijke snelheid is de helling van de raaklijn. Welke twee punten van de raaklijn ken je?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & (t_1; x_1) = (0 \, \mathrm{s}; 4{,}0 \, \mathrm{m}) & & (t_2; x_2) = (3{,}0 \, \mathrm{s}; 16 \, \mathrm{m})
            \end{align*}

            \underline{Gevraagd:} $v_x(2{,}0 \, \mathrm{s})$\\

            \underline{Oplossing:}\\
            \[
                v_x = \frac{x_2 - x_1}{t_2 - t_1} = \frac{16 \, \mathrm{m} - 4{,}0 \, \mathrm{m}}{3{,}0 \, \mathrm{s} - 0 \, \mathrm{s}} = 4{,}0 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde snelheid $\langle v_x \rangle$ tussen $t = 0 \, \mathrm{s}$ en $t = 2{,}0 \, \mathrm{s}$. Is de snelheid op $t = 0 \, \mathrm{s}$ groter of kleiner dan deze gemiddelde snelheid?

        \[ \langle v_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{6.0} \, \mathrm{m/s} \]

        De snelheid op $t = 0 \, \mathrm{s}$ is: \wordChoice{
            \choice[correct]{groter}
            \choice{kleiner}
        }

        \begin{hint}
            Hoe verandert de helling van de grafiek tussen $t = 0 \, \mathrm{s}$ en $t = 2{,}0 \, \mathrm{s}$? Vergelijk het begin met het einde van het interval.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(0) = 0 \, \mathrm{m} & & x(2{,}0 \, \mathrm{s}) = 12 \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $\langle v_x \rangle$\\

            \underline{Oplossing:}\\
            \[
                \langle v_x \rangle = \frac{\Delta x}{\Delta t} = \frac{12 \, \mathrm{m} - 0 \, \mathrm{m}}{2{,}0 \, \mathrm{s} - 0 \, \mathrm{s}} = 6{,}0 \, \mathrm{m/s}
            \]
            De grafiek wordt vlakker naarmate $t$ toeneemt: de helling (en dus de snelheid) neemt af. Op $t = 2{,}0 \, \mathrm{s}$ is de snelheid $4{,}0 \, \mathrm{m/s}$, dus op $t = 0 \, \mathrm{s}$ is ze groter dan de gemiddelde $6{,}0 \, \mathrm{m/s}$.
        \end{solution}
    \end{question}

    \begin{question}
        Wat is de snelheid op $t = 4{,}0 \, \mathrm{s}$, het hoogste punt van de grafiek, en wat is het teken van de versnelling $a_x$ tijdens de hele beweging?

        De snelheid $v_x(4{,}0 \, \mathrm{s})$ is: \wordChoice{
            \choice{positief}
            \choice{negatief}
            \choice[correct]{nul}
        }

        De versnelling $a_x$ is: \wordChoice{
            \choice{positief}
            \choice[correct]{negatief}
            \choice{nul}
        }

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = -1{,}0 \, \mathrm{m/s^2} \cdot t^{2} + 8{,}0 \, \mathrm{m/s} \cdot t$\\

            \underline{Gevraagd:} $v_x(4{,}0 \, \mathrm{s})$ en het teken van $a_x$\\

            \underline{Oplossing:}\\
            In het hoogste punt van de grafiek is de raaklijn horizontaal: de helling is nul, dus $v_x = 0 \, \mathrm{m/s}$. Via de afgeleide:
            \[
                v_x(t) = x'(t) = -2{,}0 \, \mathrm{m/s^2} \cdot t + 8{,}0 \, \mathrm{m/s} \quad \Rightarrow \quad v_x(4{,}0 \, \mathrm{s}) = -8{,}0 \, \mathrm{m/s} + 8{,}0 \, \mathrm{m/s} = 0 \, \mathrm{m/s}
            \]
            De versnelling is de afgeleide van de snelheid: $a_x = -2{,}0 \, \mathrm{m/s^2}$. Ze is negatief en constant: de snelheid wordt voortdurend kleiner.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken met behulp van de positievergelijking de snelheid $v_x$ op $t = 6{,}0 \, \mathrm{s}$.

        \[ v_x(6{,}0 \, \mathrm{s}) = \answer[tolerance=0.01,onlineshowanswerbutton]{-4.0} \, \mathrm{m/s} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = -2{,}0 \, \mathrm{m/s^2} \cdot t + 8{,}0 \, \mathrm{m/s}$\\

            \underline{Gevraagd:} $v_x(6{,}0 \, \mathrm{s})$\\

            \underline{Oplossing:}\\
            \[
                v_x(6{,}0 \, \mathrm{s}) = -2{,}0 \, \mathrm{m/s^2} \cdot 6{,}0 \, \mathrm{s} + 8{,}0 \, \mathrm{m/s} = -4{,}0 \, \mathrm{m/s}
            \]
            Na $t = 4{,}0 \, \mathrm{s}$ beweegt het voorwerp dus in de negatieve zin, wat overeenkomt met de dalende grafiek.
        \end{solution}
    \end{question}
\end{exercise}

\section{Grafieken: van de $v_x(t)$- en $a_x(t)$-grafiek naar snelheid en positie}
De helling van een $v_x(t)$-grafiek is de versnelling. De oppervlakte tussen de grafiek en de $t$-as (driehoeken en rechthoeken) geeft de verandering van de positie, of bij een $a_x(t)$-grafiek de verandering van de snelheid.

%———————————————————————————————————————
% OEFENING 2: v(t)-grafiek met zinsomkering
\begin{exercise}
    Een radiografisch bestuurbaar wagentje rijdt over een rechte rail langs de $x$-as. Op $t = 0{,}0 \, \mathrm{s}$ staat het op $x = +15 \, \mathrm{m}$. De figuur toont de snelheid $v_x$ als functie van de tijd.

    \begin{center}
        \begin{tikzpicture}[x=0.5cm,y=0.3cm]
            \draw[gray!40, very thin] (0,-8) grid[xstep=2,ystep=4] (20,8);
            \draw[positie] (0,0) -- (21,0) node[right] {$t\,(\mathrm{s})$};
            \draw[positie] (0,-9) -- (0,9.5) node[above] {$v_x\,(\mathrm{m/s})$};
            \foreach \tx in {2,4,...,20} \draw (\tx,0.3) -- (\tx,-0.3) node[below=3pt, font=\small] {\tx};
            \foreach \vy in {-8,-4,4,8} \draw (0.2,\vy) -- (-0.2,\vy) node[left, font=\small] {$\vy$};
            \draw[line width=1pt, red!100!black] (0,0) -- (4,8) -- (10,8) -- (14,-8) -- (18,-8) -- (20,0);
        \end{tikzpicture}
    \end{center}

    \begin{question}
        Bereken de versnelling $a_x$ tussen $t = 10 \, \mathrm{s}$ en $t = 14 \, \mathrm{s}$.

        \[ a_x = \answer[tolerance=0.01,onlineshowanswerbutton]{-4.0} \, \mathrm{m/s^2} \]

        Op $t = 13 \, \mathrm{s}$ geldt: \wordChoice{
            \choice[correct]{de grootte van de snelheid neemt toe}
            \choice{de grootte van de snelheid neemt af}
        }

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_x(10 \, \mathrm{s}) = 8{,}0 \, \mathrm{m/s} & & v_x(14 \, \mathrm{s}) = -8{,}0 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} $a_x$\\

            \underline{Oplossing:}\\
            De grafiek is in dit interval een rechte, dus de versnelling is constant en gelijk aan de helling:
            \[
                a_x = \frac{\Delta v_x}{\Delta t} = \frac{-8{,}0 \, \mathrm{m/s} - 8{,}0 \, \mathrm{m/s}}{14 \, \mathrm{s} - 10 \, \mathrm{s}} = -4{,}0 \, \mathrm{m/s^2}
            \]
            Op $t = 13 \, \mathrm{s}$ is de snelheid negatief ($v_x < 0$) en de versnelling ook negatief ($a_x < 0$). Ze hebben hetzelfde teken, dus de snelheid wordt (in grootte) groter: het wagentje versnelt in de negatieve zin.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de verplaatsing $\Delta x$ en de afgelegde weg $s$ tussen $t = 0 \, \mathrm{s}$ en $t = 20 \, \mathrm{s}$.

        \[ \Delta x = \answer[tolerance=0.01,onlineshowanswerbutton]{24} \, \mathrm{m} \qquad \text{en} \qquad s = \answer[tolerance=0.01,onlineshowanswerbutton]{120} \, \mathrm{m} \]

        \begin{hint}
            Wat stelt de oppervlakte tussen de grafiek en de $t$-as voor? Wat gebeurt er met het teken van die oppervlakte wanneer de snelheid negatief is?
        \end{hint}
        \begin{hint}
            Splits de grafiek op in driehoeken en rechthoeken. Op welk tijdstip verandert de zin van de beweging?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            Rechtlijnige beweging; de snelheid wordt afgelezen uit de $v_x(t)$-grafiek. De zin van de beweging keert om op $t = 12 \, \mathrm{s}$ (daar is $v_x = 0$).\\

            \underline{Gevraagd:} $\Delta x$ en $s$\\

            \underline{Oplossing:}\\
            De verplaatsing is de oppervlakte onder de grafiek, met een negatief teken voor het deel onder de $t$-as. We splitsen op in zes stukken:
            \begin{align*}
                A_1 &= \tfrac{1}{2} \cdot 4{,}0 \, \mathrm{s} \cdot 8{,}0 \, \mathrm{m/s} = +16 \, \mathrm{m} & & (0 \, \mathrm{s} \text{ tot } 4 \, \mathrm{s})\\
                A_2 &= 6{,}0 \, \mathrm{s} \cdot 8{,}0 \, \mathrm{m/s} = +48 \, \mathrm{m} & & (4 \, \mathrm{s} \text{ tot } 10 \, \mathrm{s})\\
                A_3 &= \tfrac{1}{2} \cdot 2{,}0 \, \mathrm{s} \cdot 8{,}0 \, \mathrm{m/s} = +8{,}0 \, \mathrm{m} & & (10 \, \mathrm{s} \text{ tot } 12 \, \mathrm{s})\\
                A_4 &= \tfrac{1}{2} \cdot 2{,}0 \, \mathrm{s} \cdot (-8{,}0 \, \mathrm{m/s}) = -8{,}0 \, \mathrm{m} & & (12 \, \mathrm{s} \text{ tot } 14 \, \mathrm{s})\\
                A_5 &= 4{,}0 \, \mathrm{s} \cdot (-8{,}0 \, \mathrm{m/s}) = -32 \, \mathrm{m} & & (14 \, \mathrm{s} \text{ tot } 18 \, \mathrm{s})\\
                A_6 &= \tfrac{1}{2} \cdot 2{,}0 \, \mathrm{s} \cdot (-8{,}0 \, \mathrm{m/s}) = -8{,}0 \, \mathrm{m} & & (18 \, \mathrm{s} \text{ tot } 20 \, \mathrm{s})
            \end{align*}
            De verplaatsing is de som:
            \[
                \Delta x = 16 \, \mathrm{m} + 48 \, \mathrm{m} + 8{,}0 \, \mathrm{m} - 8{,}0 \, \mathrm{m} - 32 \, \mathrm{m} - 8{,}0 \, \mathrm{m} = 24 \, \mathrm{m}
            \]
            Voor de afgelegde weg tellen we alle oppervlaktes als positief:
            \[
                s = 16 \, \mathrm{m} + 48 \, \mathrm{m} + 8{,}0 \, \mathrm{m} + 8{,}0 \, \mathrm{m} + 32 \, \mathrm{m} + 8{,}0 \, \mathrm{m} = 120 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde snelheid $\langle v_x \rangle$ tussen $t = 0 \, \mathrm{s}$ en $t = 20 \, \mathrm{s}$.

        \[ \langle v_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{1.2} \, \mathrm{m/s} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & \Delta x = 24 \, \mathrm{m} & & \Delta t = 20 \, \mathrm{s} - 0 \, \mathrm{s} = 20 \, \mathrm{s}
            \end{align*}

            \underline{Gevraagd:} $\langle v_x \rangle$\\

            \underline{Oplossing:}\\
            \[
                \langle v_x \rangle = \frac{\Delta x}{\Delta t} = \frac{24 \, \mathrm{m}}{20 \, \mathrm{s}} = 1{,}2 \, \mathrm{m/s}
            \]
            Het is dus de verplaatsing (en niet de afgelegde weg) die in de gemiddelde snelheid terechtkomt.
        \end{solution}
    \end{question}

    \begin{question}
        Wat is de grootste positie $x_{\max}$ die het wagentje tijdens de rit bereikt?

        \begin{hint}
            Op welk tijdstip stopt het wagentje met vooruitrijden? Vergeet de beginpositie niet.
        \end{hint}

        \begin{solution}\nl

            \end{align*}

            \]
        \end{solution}
    \end{question}

    \begin{question}
        Het wagentje passeert de positie $x = 55 \, \mathrm{m}$ twee keer. Op welke tijdstippen $t_1$ en $t_2$ (met $t_1 < t_2$)?

        \[ t_1 = \answer[tolerance=0.01,onlineshowanswerbutton]{7.0} \, \mathrm{s} \qquad \text{en} \qquad t_2 = \answer[tolerance=0.01,onlineshowanswerbutton]{17} \, \mathrm{s} \]

        \begin{hint}
            Hoeveel moet het wagentje verplaatst zijn ten opzichte van zijn beginpositie om op $55 \, \mathrm{m}$ te staan? In welke tijdsintervallen kan dat gebeuren?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(0) = 15 \, \mathrm{m} & & x = 55 \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $t_1$ en $t_2$\\

            \underline{Oplossing:}\\
            Het wagentje moet $55 \, \mathrm{m} - 15 \, \mathrm{m} = 40 \, \mathrm{m}$ verplaatst zijn ten opzichte van $t = 0 \, \mathrm{s}$.

            De verplaatsing na $4{,}0 \, \mathrm{s}$ is $16 \, \mathrm{m}$ en na $10 \, \mathrm{s}$ is dat $64 \, \mathrm{m}$. De eerste keer gebeurt dit dus tijdens de constante snelheid van $8{,}0 \, \mathrm{m/s}$:
            \[
                16 \, \mathrm{m} + 8{,}0 \, \mathrm{m/s} \cdot (t_1 - 4{,}0 \, \mathrm{s}) = 40 \, \mathrm{m} \quad \Rightarrow \quad t_1 = 4{,}0 \, \mathrm{s} + \frac{24 \, \mathrm{m}}{8{,}0 \, \mathrm{m/s}} = 7{,}0 \, \mathrm{s}

            \[
                64 \, \mathrm{m} + (-8{,}0 \, \mathrm{m/s}) \cdot (t_2 - 14 \, \mathrm{s}) = 40 \, \mathrm{m} \quad \Rightarrow \quad t_2 = 14 \, \mathrm{s} + \frac{24 \, \mathrm{m}}{8{,}0 \, \mathrm{m/s}} = 17 \, \mathrm{s}
            \]
        \end{solution}
    \end{question}
\end{exercise}

\begin{exercise}
    Een voorwerp beweegt langs de $x$-as. Op $t = 0{,}0 \, \mathrm{s}$ bevindt het zich in de oorsprong en heeft het een snelheid $v_x = -6{,}0 \, \mathrm{m/s}$. De figuur toont de versnelling $a_x$ als functie van de tijd.

    \begin{center}
        \begin{tikzpicture}[x=0.6cm,y=0.5cm]
            \draw[gray!40, very thin] (0,-2) grid[xstep=1,ystep=1] (10,3);
            \draw[positie] (0,0) -- (11,0) node[right] {$t\,(\mathrm{s})$};
            \draw[positie] (0,-2.7) -- (0,3.8) node[above] {$a_x\,(\mathrm{m/s^2})$};
            \foreach \tx in {2,4,...,10} \draw (\tx,0.1) -- (\tx,-0.1) node[below=3pt, font=\small] {\tx};
            \foreach \ay in {-2,-1,1,2,3} \draw (0.1,\ay) -- (-0.1,\ay) node[left, font=\small] {$\ay$};
            \draw[line width=1pt, violet!100!black] (0,3) -- (4,3);
            \draw[line width=1pt, violet!100!black] (4,0) -- (6,0);
            \draw[line width=1pt, violet!100!black] (6,-2) -- (10,-2);
            \draw[dotted, violet!100!black] (4,3) -- (4,0) (6,0) -- (6,-2);
        \end{tikzpicture}
    \end{center}

    \begin{question}
        Bereken de snelheid $v_x$ op $t = 4{,}0 \, \mathrm{s}$.

        \[ v_x(4{,}0 \, \mathrm{s}) = \answer[tolerance=0.01,onlineshowanswerbutton]{6.0} \, \mathrm{m/s} \]

        \begin{hint}
            Wat stelt de oppervlakte onder een $a_x(t)$-grafiek voor?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_x(0) = -6{,}0 \, \mathrm{m/s} & & a_x = 3{,}0 \, \mathrm{m/s^2} \text{ voor } 0 \, \mathrm{s} \leq t \leq 4{,}0 \, \mathrm{s}
            \end{align*}

            \underline{Gevraagd:} $v_x(4{,}0 \, \mathrm{s})$\\

            \underline{Oplossing:}\\
            De oppervlakte onder de $a_x(t)$-grafiek is de snelheidsverandering:
            \[
                \Delta v_x = 3{,}0 \, \mathrm{m/s^2} \cdot 4{,}0 \, \mathrm{s} = 12 \, \mathrm{m/s}
            \]
            \[
                v_x(4{,}0 \, \mathrm{s}) = v_x(0) + \Delta v_x = -6{,}0 \, \mathrm{m/s} + 12 \, \mathrm{m/s} = 6{,}0 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Het voorwerp keert tweemaal van zin om. Op welke tijdstippen $t_1$ en $t_2$ (met $t_1 < t_2$)?

        \[ t_1 = \answer[tolerance=0.01,onlineshowanswerbutton]{2.0} \, \mathrm{s} \qquad \text{en} \qquad t_2 = \answer[tolerance=0.01,onlineshowanswerbutton]{9.0} \, \mathrm{s} \]

        \begin{hint}
            Wat is de snelheid op het moment dat het voorwerp van zin verandert?
        \end{hint}
        \begin{hint}
            Welke snelheid heeft het voorwerp op $t = 6{,}0 \, \mathrm{s}$, en hoe verandert die daarna?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_x(0) = -6{,}0 \, \mathrm{m/s} & & v_x(4{,}0 \, \mathrm{s}) = 6{,}0 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} $t_1$ en $t_2$\\

            \underline{Oplossing:}\\
            Bij een zinsomkering is $v_x = 0$. In het eerste interval ($a_x = 3{,}0 \, \mathrm{m/s^2}$) geldt:
            \[
                -6{,}0 \, \mathrm{m/s} + 3{,}0 \, \mathrm{m/s^2} \cdot t_1 = 0 \quad \Rightarrow \quad t_1 = \frac{6{,}0 \, \mathrm{m/s}}{3{,}0 \, \mathrm{m/s^2}} = 2{,}0 \, \mathrm{s}
            \]
            Tussen $4{,}0 \, \mathrm{s}$ en $6{,}0 \, \mathrm{s}$ is $a_x = 0$, dus blijft de snelheid $6{,}0 \, \mathrm{m/s}$. Vanaf $t = 6{,}0 \, \mathrm{s}$ is $a_x = -2{,}0 \, \mathrm{m/s^2}$:
            \[
                6{,}0 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot (t_2 - 6{,}0 \, \mathrm{s}) = 0 \quad \Rightarrow \quad t_2 = 6{,}0 \, \mathrm{s} + \frac{6{,}0 \, \mathrm{m/s}}{2{,}0 \, \mathrm{m/s^2}} = 9{,}0 \, \mathrm{s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de verplaatsing $\Delta x$ en de afgelegde weg $s$ tussen $t = 0 \, \mathrm{s}$ en $t = 10 \, \mathrm{s}$.

        \[ \Delta x = \answer[tolerance=0.01,onlineshowanswerbutton]{20} \, \mathrm{m} \qquad \text{en} \qquad s = \answer[tolerance=0.01,onlineshowanswerbutton]{34} \, \mathrm{m} \]

        \begin{hint}
            Schets eerst de $v_x(t)$-grafiek. Gebruik de snelheden op $t = 0$, $2{,}0$, $4{,}0$, $6{,}0$, $9{,}0$ en $10 \, \mathrm{s}$.
        \end{hint}
        \begin{hint}
            De oppervlakte onder de $v_x(t)$-grafiek is de verplaatsing. Rekening houden met het teken geeft de verplaatsing, alles positief tellen geeft de afgelegde weg.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_x(0) = -6{,}0 \, \mathrm{m/s} & & v_x(2{,}0 \, \mathrm{s}) = 0 \, \mathrm{m/s}\\
                & v_x(4{,}0 \, \mathrm{s}) = v_x(6{,}0 \, \mathrm{s}) = 6{,}0 \, \mathrm{m/s} & & v_x(9{,}0 \, \mathrm{s}) = 0 \, \mathrm{m/s}\\
                & v_x(10 \, \mathrm{s}) = -2{,}0 \, \mathrm{m/s} & &
            \end{align*}

            \underline{Gevraagd:} $\Delta x$ en $s$\\

            \underline{Oplossing:}\\
            De $v_x(t)$-grafiek bestaat uit rechte stukken. De oppervlaktes zijn:
            \begin{align*}
                A_1 &= \tfrac{1}{2} \cdot 2{,}0 \, \mathrm{s} \cdot (-6{,}0 \, \mathrm{m/s}) = -6{,}0 \, \mathrm{m} & & (0 \, \mathrm{s} \text{ tot } 2{,}0 \, \mathrm{s})\\
                A_2 &= \tfrac{1}{2} \cdot 2{,}0 \, \mathrm{s} \cdot 6{,}0 \, \mathrm{m/s} = +6{,}0 \, \mathrm{m} & & (2{,}0 \, \mathrm{s} \text{ tot } 4{,}0 \, \mathrm{s})\\

            \]

            \[
                s = 6{,}0 \, \mathrm{m} + 6{,}0 \, \mathrm{m} + 12 \, \mathrm{m} + 9{,}0 \, \mathrm{m} + 1{,}0 \, \mathrm{m} = 34 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}

% OEFENING: twee v(t)-grafieken, inhalen
\begin{exercise}
    Twee wagentjes $A$ en $B$ vertrekken op $t = 0 \, \mathrm{s}$ samen vanuit de oorsprong en rijden in dezelfde zin langs de $x$-as. De figuur toont hun snelheden $v_x$ als functie van de tijd. Wagentje $A$ vertrekt uit rust.

    \begin{center}
        \begin{tikzpicture}[x=0.55cm,y=0.2cm]
            \draw[gray!40, very thin] (0,0) grid[xstep=1,ystep=4] (10,20);
            \draw[positie] (0,0) -- (10.8,0) node[right] {$t\,(\mathrm{s})$};
            \draw[positie] (0,0) -- (0,23) node[above] {$v_x\,(\mathrm{m/s})$};
            \foreach \tx in {2,4,...,10} \draw (\tx,0.4) -- (\tx,-0.4) node[below=3pt, font=\small] {\tx};
            \foreach \vy in {4,8,12,16,20} \draw (0.1,\vy) -- (-0.1,\vy) node[left, font=\small] {$\vy$};
            \draw[line width=1pt, red!100!black] (0,0) -- (10,20) node[above left] {$A$};
            \draw[line width=1pt, blue!70!black] (0,8) -- (10,8) node[above left] {$B$};
        \end{tikzpicture}
    \end{center}

    \begin{question}
        Bereken de versnelling van wagentje $A$.

        \[ a_A = \answer[tolerance=0.01,onlineshowanswerbutton]{2.0} \, \mathrm{m/s^2} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_A(0) = 0 \, \mathrm{m/s} & & v_A(10 \, \mathrm{s}) = 20 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} $a_A$\\

            \underline{Oplossing:}\\
            \[
                a_A = \frac{\Delta v_x}{\Delta t} = \frac{20 \, \mathrm{m/s} - 0 \, \mathrm{m/s}}{10 \, \mathrm{s} - 0 \, \mathrm{s}} = 2{,}0 \, \mathrm{m/s^2}
            \]
            Wagentje $B$ rijdt met constante snelheid $8{,}0 \, \mathrm{m/s}$, dus $a_B = 0 \, \mathrm{m/s^2}$.
        \end{solution}
    \end{question}

    \begin{question}
        Op welk tijdstip hebben beide wagentjes dezelfde snelheid?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{4.0} \, \mathrm{s} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_A(t) = 2{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_B = 8{,}0 \, \mathrm{m/s}$\\

            \underline{Gevraagd:} $t$ waarvoor $v_A = v_B$\\

            \underline{Oplossing:}\\
            \[
                2{,}0 \, \mathrm{m/s^2} \cdot t = 8{,}0 \, \mathrm{m/s} \quad \Rightarrow \quad t = 4{,}0 \, \mathrm{s}
            \]
            Dit is het snijpunt van beide grafieken.
        \end{solution}
    \end{question}

    \begin{question}
        Wat is de grootste afstand tussen beide wagentjes tijdens de eerste $10 \, \mathrm{s}$? Wagentje $B$ ligt dan voor.

        \[ \answer[tolerance=0.01,onlineshowanswerbutton]{16} \, \mathrm{m} \]

        \begin{hint}
            Vergelijk hoe ver elk wagentje is gekomen op het moment dat ze dezelfde snelheid hebben. Hoe bepaal je een afgelegde afstand uit een $v_x(t)$-grafiek?
        \end{hint}
        \begin{hint}
            Voor $t < 4{,}0 \, \mathrm{s}$ is $B$ sneller dan $A$, daarna is $A$ sneller. Wanneer is de afstand tussen beide dus het grootst?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & t = 4{,}0 \, \mathrm{s} & & v_A(4{,}0 \, \mathrm{s}) = v_B = 8{,}0 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} maximale afstand tussen $A$ en $B$\\

            \underline{Oplossing:}\\
            Voor $t < 4{,}0 \, \mathrm{s}$ rijdt $B$ sneller dan $A$ en groeit de afstand. Daarna rijdt $A$ sneller en wordt de afstand kleiner. De afstand is dus maximaal op $t = 4{,}0 \, \mathrm{s}$.

            De afgelegde afstand is de oppervlakte onder de $v_x(t)$-grafiek (een rechthoek voor $B$, een driehoek voor $A$):
            \begin{align*}
                \Delta x_B &= 4{,}0 \, \mathrm{s} \cdot 8{,}0 \, \mathrm{m/s} = 32 \, \mathrm{m}\\
                \Delta x_A &= \tfrac{1}{2} \cdot 4{,}0 \, \mathrm{s} \cdot 8{,}0 \, \mathrm{m/s} = 16 \, \mathrm{m}
            \end{align*}
            De afstand is $32 \, \mathrm{m} - 16 \, \mathrm{m} = 16 \, \mathrm{m}$.
        \end{solution}
    \end{question}

    \begin{question}
        Op welk tijdstip haalt $A$ wagentje $B$ in, en op welke positie $x$ is dat?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{8.0} \, \mathrm{s} \qquad \text{en} \qquad x = \answer[tolerance=0.01,onlineshowanswerbutton]{64} \, \mathrm{m} \]

        \begin{hint}
            Op het moment van inhalen hebben beide wagentjes evenveel afstand afgelegd. Druk de oppervlakte onder elke grafiek uit in functie van $t$.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_A(t) = 2{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_B = 8{,}0 \, \mathrm{m/s}$\\

            \underline{Gevraagd:} $t$ en $x$ waarvoor $\Delta x_A = \Delta x_B$\\

            \underline{Oplossing:}\\
            Na een tijd $t$ is de afgelegde afstand van $B$ een rechthoek en die van $A$ een driehoek met hoogte $v_A(t) = 2{,}0 \, \mathrm{m/s^2} \cdot t$:
            \begin{align*}
                \Delta x_B &= 8{,}0 \, \mathrm{m/s} \cdot t\\
                \Delta x_A &= \tfrac{1}{2} \cdot t \cdot 2{,}0 \, \mathrm{m/s^2} \cdot t = 1{,}0 \, \mathrm{m/s^2} \cdot t^{2}
            \end{align*}
            We stellen beide gelijk (eenheden even weggelaten, $t$ in $\mathrm{s}$): $t^{2} = 8t$, dus $t = 8{,}0 \, \mathrm{s}$ (naast de oplossing $t = 0$ van het vertrek). De positie is:
            \[
                x = 8{,}0 \, \mathrm{m/s} \cdot 8{,}0 \, \mathrm{s} = 64 \, \mathrm{m}
            \]
            Op dat moment rijdt $A$ met $16 \, \mathrm{m/s}$ en $B$ met $8{,}0 \, \mathrm{m/s}$: $A$ haalt $B$ in.
        \end{solution}
    \end{question}
\end{exercise}

\section{Functievoorschriften: van snelheid en positie naar versnelling}
Met afgeleiden ga je van $x(t)$ naar $v_x(t)$ en $a_x(t)$. Hoe je van een versnelling terug naar snelheid en positie gaat, wordt hier niet gevraagd.

% OEFENING: v(t)-voorschrift naar a(t)
\begin{exercise}
    De snelheid van een voorwerp langs de $x$-as wordt gegeven door
    \[
        v_x(t) = 3{,}0 \, \mathrm{\frac{m}{s^3}} \cdot t^{2} - 12 \, \mathrm{\frac{m}{s^2}} \cdot t
    \]
    met $t$ in seconden.

    \begin{question}
        Op welk tijdstip $t > 0$ staat het voorwerp even stil, en hoe groot is de versnelling $a_x$ op dat moment?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{4.0} \, \mathrm{s} \qquad \text{en} \qquad a_x = \answer[tolerance=0.01,onlineshowanswerbutton]{12} \, \mathrm{m/s^2} \]

        \begin{hint}
            Hoe vind je de versnellingsvergelijking vanuit de snelheidsvergelijking?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 3{,}0 \, \mathrm{m/s^3} \cdot t^{2} - 12 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $t$ waarvoor $v_x = 0$ en $a_x$ op dat moment\\

            \underline{Oplossing:}\\
            Stilstaan betekent $v_x = 0$. We zetten $t$ buiten haakjes:
            \[
                3{,}0 \, \mathrm{m/s^3} \cdot t \cdot (t - 4{,}0 \, \mathrm{s}) = 0 \quad \Rightarrow \quad t = 0 \, \mathrm{s} \text{ of } t = 4{,}0 \, \mathrm{s}
            \]
            De versnelling is de afgeleide van de snelheid:
            \[
                a_x(t) = v_x'(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t - 12 \, \mathrm{m/s^2}
            \]
            \[
                a_x(4{,}0 \, \mathrm{s}) = 6{,}0 \, \mathrm{m/s^3} \cdot 4{,}0 \, \mathrm{s} - 12 \, \mathrm{m/s^2} = 12 \, \mathrm{m/s^2}
            \]
            Een voorwerp dat even stilstaat, heeft dus niet noodzakelijk een versnelling nul.
        \end{solution}
    \end{question}

    \begin{question}
        Wat is de kleinste snelheid die het voorwerp bereikt?

        \[ v_{x,\min} = \answer[tolerance=0.01,onlineshowanswerbutton]{-12} \, \mathrm{m/s} \]

        \begin{hint}
            Bij een grafiek met een dal of een top is de helling nul. Wat betekent dat voor de versnelling?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $a_x(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t - 12 \, \mathrm{m/s^2}$\\

            \underline{Gevraagd:} $v_{x,\min}$\\

            \underline{Oplossing:}\\
            De snelheid is extreem als haar afgeleide, de versnelling, nul is:
            \[
                6{,}0 \, \mathrm{m/s^3} \cdot t - 12 \, \mathrm{m/s^2} = 0 \quad \Rightarrow \quad t = 2{,}0 \, \mathrm{s}
            \]
            \begin{align*}
                v_x(2{,}0 \, \mathrm{s}) &= 3{,}0 \, \mathrm{m/s^3} \cdot (2{,}0 \, \mathrm{s})^{2} - 12 \, \mathrm{m/s^2} \cdot 2{,}0 \, \mathrm{s}\\
                &= 12 \, \mathrm{m/s} - 24 \, \mathrm{m/s} = -12 \, \mathrm{m/s}
            \end{align*}
            De grafiek van $v_x(t)$ is een dalparabool, dus dit is een minimum.
        \end{solution}
    \end{question}

    \begin{question}
        Op $t = 3{,}0 \, \mathrm{s}$ geldt: \wordChoice{
            \choice{de grootte van de snelheid neemt toe}
            \choice[correct]{de grootte van de snelheid neemt af}
        }

        \begin{hint}
            Bepaal het teken van de snelheid en het teken van de versnelling op dat moment en vergelijk ze.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 3{,}0 \, \mathrm{m/s^3} \cdot t^{2} - 12 \, \mathrm{m/s^2} \cdot t$ en $a_x(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t - 12 \, \mathrm{m/s^2}$\\

            \underline{Gevraagd:} evolutie van de grootte van de snelheid op $t = 3{,}0 \, \mathrm{s}$\\

            \underline{Oplossing:}\\
            \begin{align*}
                v_x(3{,}0 \, \mathrm{s}) &= 27 \, \mathrm{m/s} - 36 \, \mathrm{m/s} = -9{,}0 \, \mathrm{m/s}\\
                a_x(3{,}0 \, \mathrm{s}) &= 18 \, \mathrm{m/s^2} - 12 \, \mathrm{m/s^2} = +6{,}0 \, \mathrm{m/s^2}
            \end{align*}
            De snelheid is negatief en de versnelling positief: ze hebben een verschillend teken. De grootte van de snelheid neemt dus af (het voorwerp remt af terwijl het in de negatieve zin beweegt).
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING 4: positievergelijking (derde graad), afgeleiden en zinsomkering
\begin{exercise}
    De positie van een voorwerp op de $x$-as wordt gegeven door
    \[
        x(t) = 2{,}0 \, \mathrm{\frac{m}{s^3}} \cdot t^{3} - 9{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t^{2} + 12 \, \mathrm{\frac{m}{s}} \cdot t + 1{,}0 \, \mathrm{m}
    \]
    met $t$ in seconden. We bekijken de beweging tussen $t = 0 \, \mathrm{s}$ en $t = 3{,}0 \, \mathrm{s}$.

    \begin{question}
        Het voorwerp staat twee keer even stil. Op welke tijdstippen $t_1$ en $t_2$ (met $t_1 < t_2$)?

        \[ t_1 = \answer[tolerance=0.01,onlineshowanswerbutton]{1.0} \, \mathrm{s} \qquad \text{en} \qquad t_2 = \answer[tolerance=0.01,onlineshowanswerbutton]{2.0} \, \mathrm{s} \]

        \begin{hint}
            Welke grootheid is nul als een voorwerp even stilstaat, en hoe vind je de vergelijking van die grootheid vanuit $x(t)$?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = 2{,}0 \, \mathrm{m/s^3} \cdot t^{3} - 9{,}0 \, \mathrm{m/s^2} \cdot t^{2} + 12 \, \mathrm{m/s} \cdot t + 1{,}0 \, \mathrm{m}$\\

            \underline{Gevraagd:} de tijdstippen waarvoor $v_x = 0$\\

            \underline{Oplossing:}\\
            De snelheid is de afgeleide van de positie:
            \[
                v_x(t) = x'(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t^{2} - 18 \, \mathrm{m/s^2} \cdot t + 12 \, \mathrm{m/s}
            \]
            Stilstaan betekent $v_x = 0$. We laten de eenheden even weg ($t$ in $\mathrm{s}$) en delen door $6$:
            \[
                t^{2} - 3t + 2 = 0 \quad \Rightarrow \quad t = \frac{3 \pm \sqrt{9 - 8}}{2} \quad \Rightarrow \quad t_1 = 1{,}0 \, \mathrm{s} \text{ en } t_2 = 2{,}0 \, \mathrm{s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de verplaatsing $\Delta x$ tussen $t = 0 \, \mathrm{s}$ en $t = 3{,}0 \, \mathrm{s}$.

        \[ \Delta x = \answer[tolerance=0.01,onlineshowanswerbutton]{9.0} \, \mathrm{m} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = 2{,}0 \, \mathrm{m/s^3} \cdot t^{3} - 9{,}0 \, \mathrm{m/s^2} \cdot t^{2} + 12 \, \mathrm{m/s} \cdot t + 1{,}0 \, \mathrm{m}$\\

            \underline{Gevraagd:} $\Delta x$\\

            \underline{Oplossing:}\\
            We berekenen de positie op beide tijdstippen:
            \begin{align*}
                x(0) &= 1{,}0 \, \mathrm{m}\\
                x(3{,}0 \, \mathrm{s}) &= 2{,}0 \, \mathrm{m/s^3} \cdot (3{,}0 \, \mathrm{s})^{3} - 9{,}0 \, \mathrm{m/s^2} \cdot (3{,}0 \, \mathrm{s})^{2} + 12 \, \mathrm{m/s} \cdot 3{,}0 \, \mathrm{s} + 1{,}0 \, \mathrm{m}\\
                &= 54 \, \mathrm{m} - 81 \, \mathrm{m} + 36 \, \mathrm{m} + 1{,}0 \, \mathrm{m} = 10 \, \mathrm{m}
            \end{align*}
            \[
                \Delta x = x(3{,}0 \, \mathrm{s}) - x(0) = 10 \, \mathrm{m} - 1{,}0 \, \mathrm{m} = 9{,}0 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de afgelegde weg $s$ tussen $t = 0 \, \mathrm{s}$ en $t = 3{,}0 \, \mathrm{s}$.

        \[ s = \answer[tolerance=0.01,onlineshowanswerbutton]{11} \, \mathrm{m} \]

        \begin{hint}
            De afgelegde weg is niet gelijk aan de verplaatsing als het voorwerp van zin verandert. Splits het tijdsinterval op in stukken waarin het voorwerp telkens in dezelfde zin beweegt.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            Zinsomkeringen op $t_1 = 1{,}0 \, \mathrm{s}$ en $t_2 = 2{,}0 \, \mathrm{s}$; $x(0) = 1{,}0 \, \mathrm{m}$ en $x(3{,}0 \, \mathrm{s}) = 10 \, \mathrm{m}$.\\

            \underline{Gevraagd:} $s$\\

            \underline{Oplossing:}\\
            We berekenen de posities op de momenten van zinsomkering:
            \begin{align*}
                x(1{,}0 \, \mathrm{s}) &= 2{,}0 \, \mathrm{m} - 9{,}0 \, \mathrm{m} + 12 \, \mathrm{m} + 1{,}0 \, \mathrm{m} = 6{,}0 \, \mathrm{m}\\
                x(2{,}0 \, \mathrm{s}) &= 16 \, \mathrm{m} - 36 \, \mathrm{m} + 24 \, \mathrm{m} + 1{,}0 \, \mathrm{m} = 5{,}0 \, \mathrm{m}
            \end{align*}
            Het voorwerp gaat dus van $1{,}0 \, \mathrm{m}$ naar $6{,}0 \, \mathrm{m}$, dan terug naar $5{,}0 \, \mathrm{m}$ en daarna naar $10 \, \mathrm{m}$:
            \[
                s = \lvert 6{,}0 \, \mathrm{m} - 1{,}0 \, \mathrm{m} \rvert + \lvert 5{,}0 \, \mathrm{m} - 6{,}0 \, \mathrm{m} \rvert + \lvert 10 \, \mathrm{m} - 5{,}0 \, \mathrm{m} \rvert = 5{,}0 \, \mathrm{m} + 1{,}0 \, \mathrm{m} + 5{,}0 \, \mathrm{m} = 11 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Wat is de snelheid $v_x$ van het voorwerp op het moment dat zijn versnelling nul is?

        \[ v_x = \answer[tolerance=0.01,onlineshowanswerbutton]{-1.5} \, \mathrm{m/s} \]

        \begin{hint}
            Hoe vind je de versnellingsvergelijking $a_x(t)$ vanuit $x(t)$? Op welk tijdstip is die nul?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t^{2} - 18 \, \mathrm{m/s^2} \cdot t + 12 \, \mathrm{m/s}$\\

            \underline{Gevraagd:} $v_x$ als $a_x = 0$\\

            \underline{Oplossing:}\\
            De versnelling is de afgeleide van de snelheid:
            \[
                a_x(t) = v_x'(t) = 12 \, \mathrm{m/s^3} \cdot t - 18 \, \mathrm{m/s^2}
            \]
            Dit is nul voor $t = \dfrac{18 \, \mathrm{m/s^2}}{12 \, \mathrm{m/s^3}} = 1{,}5 \, \mathrm{s}$. De snelheid op dat moment is:
            \begin{align*}
                v_x(1{,}5 \, \mathrm{s}) &= 6{,}0 \, \mathrm{m/s^3} \cdot (1{,}5 \, \mathrm{s})^{2} - 18 \, \mathrm{m/s^2} \cdot 1{,}5 \, \mathrm{s} + 12 \, \mathrm{m/s}\\
                &= 13{,}5 \, \mathrm{m/s} - 27 \, \mathrm{m/s} + 12 \, \mathrm{m/s} = -1{,}5 \, \mathrm{m/s}
            \end{align*}
            Het voorwerp beweegt op dat moment in de negatieve zin, en zijn snelheid is op dat moment het kleinst (meest negatief).
        \end{solution}
    \end{question}

    \begin{question}
        Op twee tijdstippen tussen $0 \, \mathrm{s}$ en $3{,}0 \, \mathrm{s}$ is de ogenblikkelijke snelheid exact gelijk aan de gemiddelde snelheid over dat interval. Bepaal die tijdstippen $t_1$ en $t_2$ (met $t_1 < t_2$).

        \[ t_1 = \answer[tolerance=0.02,onlineshowanswerbutton]{0.63} \, \mathrm{s} \qquad \text{en} \qquad t_2 = \answer[tolerance=0.02,onlineshowanswerbutton]{2.37} \, \mathrm{s} \]

        \begin{hint}
            Bereken eerst de gemiddelde snelheid over het interval. Stel daarna de snelheidsvergelijking gelijk aan die waarde.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & \Delta x = 9{,}0 \, \mathrm{m} & & \Delta t = 3{,}0 \, \mathrm{s}
            \end{align*}

            \underline{Gevraagd:} $t_1$ en $t_2$ waarvoor $v_x(t) = \langle v_x \rangle$\\

            \underline{Oplossing:}\\
            De gemiddelde snelheid is:
            \[
                \langle v_x \rangle = \frac{\Delta x}{\Delta t} = \frac{9{,}0 \, \mathrm{m}}{3{,}0 \, \mathrm{s}} = 3{,}0 \, \mathrm{m/s}
            \]
            We stellen $v_x(t) = 3{,}0 \, \mathrm{m/s}$ (eenheden even weggelaten, $t$ in $\mathrm{s}$):
            \[
                6t^{2} - 18t + 12 = 3 \quad \Rightarrow \quad 2t^{2} - 6t + 3 = 0
            \]
            \[
                t = \frac{6 \pm \sqrt{36 - 24}}{4} = \frac{6 \pm 3{,}464}{4} \quad \Rightarrow \quad t_1 = 0{,}63 \, \mathrm{s} \text{ en } t_2 = 2{,}37 \, \mathrm{s}
            \]
            Beide tijdstippen liggen binnen het interval $[0 \, \mathrm{s}; 3{,}0 \, \mathrm{s}]$.
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING 5: onbekende coëfficiënten bepalen uit voorwaarden
\begin{exercise}
    De positie van een voorwerp op de $x$-as wordt gegeven door
    \[
        x(t) = \alpha \cdot t^{3} + \beta \cdot t^{2}
    \]
    waarbij $\alpha$ en $\beta$ constanten zijn. Op $t = 0 \, \mathrm{s}$ bevindt het voorwerp zich in de oorsprong. Op $t = 2{,}0 \, \mathrm{s}$ staat het even stil op de positie $x = -8{,}0 \, \mathrm{m}$.

    \begin{question}
        Bepaal $\alpha$ en $\beta$.

        \[ \alpha = \answer[tolerance=0.01,onlineshowanswerbutton]{2.0} \, \mathrm{m/s^3} \qquad \text{en} \qquad \beta = \answer[tolerance=0.01,onlineshowanswerbutton]{-6.0} \, \mathrm{m/s^2} \]

        \begin{hint}
            Je hebt twee onbekenden, dus twee voorwaarden nodig. Welke twee gegevens uit de opgave kan je in wiskundige vorm schrijven?
        \end{hint}
        \begin{hint}
            Voor een van de voorwaarden heb je de snelheidsvergelijking nodig. Hoe vind je die?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(2{,}0 \, \mathrm{s}) = -8{,}0 \, \mathrm{m} & & v_x(2{,}0 \, \mathrm{s}) = 0 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} $\alpha$ en $\beta$\\

            \underline{Oplossing:}\\
            De snelheid is de afgeleide van de positie:
            \[
                v_x(t) = 3\alpha \cdot t^{2} + 2\beta \cdot t
            \]
            Voorwaarde 1 (even stilstaan op $t = 2{,}0 \, \mathrm{s}$), met $t$ in $\mathrm{s}$:
            \[
                v_x(2{,}0) = 12\alpha + 4\beta = 0 \quad \Rightarrow \quad \beta = -3\alpha
            \]
            Voorwaarde 2 (positie op $t = 2{,}0 \, \mathrm{s}$):
            \[
                x(2{,}0) = 8\alpha + 4\beta = -8{,}0
            \]
            We vullen $\beta = -3\alpha$ in: $8\alpha - 12\alpha = -4\alpha = -8{,}0$, dus $\alpha = 2{,}0$ en $\beta = -6{,}0$.

            Uit de eenheid van $x$ (in $\mathrm{m}$) volgt: $[\alpha] = \mathrm{m/s^3}$ en $[\beta] = \mathrm{m/s^2}$. Dus
            \[
                x(t) = 2{,}0 \, \mathrm{\frac{m}{s^3}} \cdot t^{3} - 6{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t^{2}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Na hoeveel tijd keert het voorwerp (na het vertrek) terug in de oorsprong?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{3.0} \, \mathrm{s} \]

        \begin{hint}
            Je kan de positievergelijking ontbinden in factoren.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = 2{,}0 \, \mathrm{m/s^3} \cdot t^{3} - 6{,}0 \, \mathrm{m/s^2} \cdot t^{2}$\\

            \underline{Gevraagd:} $t$ waarvoor $x = 0$ (met $t > 0$)\\

            \underline{Oplossing:}\\
            We stellen $x(t) = 0$ (eenheden even weggelaten) en zetten $2t^{2}$ buiten haakjes:
            \[
                2t^{3} - 6t^{2} = 2t^{2} \cdot (t - 3) = 0
            \]
            De oplossing $t = 0$ is het vertrek zelf. Het voorwerp is dus terug in de oorsprong op $t = 3{,}0 \, \mathrm{s}$.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de snelheid $v_x$ waarmee het voorwerp terug door de oorsprong gaat.

        \[ v_x = \answer[tolerance=0.01,onlineshowanswerbutton]{18} \, \mathrm{m/s} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t^{2} - 12 \, \mathrm{m/s^2} \cdot t$ en $t = 3{,}0 \, \mathrm{s}$\\

            \underline{Gevraagd:} $v_x(3{,}0 \, \mathrm{s})$\\

            \underline{Oplossing:}\\
            \begin{align*}
                v_x(3{,}0 \, \mathrm{s}) &= 6{,}0 \, \mathrm{m/s^3} \cdot (3{,}0 \, \mathrm{s})^{2} - 12 \, \mathrm{m/s^2} \cdot 3{,}0 \, \mathrm{s}\\
                &= 54 \, \mathrm{m/s} - 36 \, \mathrm{m/s} = 18 \, \mathrm{m/s}
            \end{align*}
            Het voorwerp gaat in de positieve zin door de oorsprong.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de afgelegde weg $s$ tussen $t = 0 \, \mathrm{s}$ en het moment waarop het voorwerp terug in de oorsprong is.

        \[ s = \answer[tolerance=0.01,onlineshowanswerbutton]{16} \, \mathrm{m} \]

        \begin{hint}
            De verplaatsing over dit interval is nul. Is de afgelegde weg dat dan ook? Bepaal wanneer het voorwerp van zin keert.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 6{,}0 \, \mathrm{m/s^3} \cdot t^{2} - 12 \, \mathrm{m/s^2} \cdot t = 6{,}0 \, \mathrm{m/s^3} \cdot t \cdot (t - 2{,}0 \, \mathrm{s})$; $x(2{,}0 \, \mathrm{s}) = -8{,}0 \, \mathrm{m}$\\

            \underline{Gevraagd:} $s$\\

            \underline{Oplossing:}\\
            De snelheid is nul op $t = 0 \, \mathrm{s}$ en $t = 2{,}0 \, \mathrm{s}$. Voor $0 \, \mathrm{s} < t < 2{,}0 \, \mathrm{s}$ is $v_x < 0$ en voor $t > 2{,}0 \, \mathrm{s}$ is $v_x > 0$. Het voorwerp beweegt dus eerst van $0 \, \mathrm{m}$ naar $-8{,}0 \, \mathrm{m}$ en keert dan terug naar $0 \, \mathrm{m}$:
            \[
                s = \lvert -8{,}0 \, \mathrm{m} - 0 \, \mathrm{m} \rvert + \lvert 0 \, \mathrm{m} - (-8{,}0 \, \mathrm{m}) \rvert = 8{,}0 \, \mathrm{m} + 8{,}0 \, \mathrm{m} = 16 \, \mathrm{m}
            \]
            Hoewel de verplaatsing en de gemiddelde snelheid nul zijn, is de afgelegde weg $16 \, \mathrm{m}$.
        \end{solution}
    \end{question}
\end{exercise}

\begin{exercise}
    Een motorrijder en een vrachtwagen rijden in dezelfde zin op dezelfde rechte weg (dezelfde $x$-as). Op $t = 0 \, \mathrm{s}$ bevindt de motor zich in de oorsprong en de vrachtwagen $27 \, \mathrm{m}$ verder. Hun posities worden gegeven door
    \begin{align*}
        x_M(t) &= 20 \, \mathrm{\frac{m}{s}} \cdot t - 1{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t^{2}\\
        x_V(t) &= 27 \, \mathrm{m} + 8{,}0 \, \mathrm{\frac{m}{s}} \cdot t
    \end{align*}
    met $t$ in seconden. Deze vergelijkingen gelden voor $0 \, \mathrm{s} \leq t \leq 10 \, \mathrm{s}$.

    \begin{question}
        Wat is de beginsnelheid van de motor in $\mathrm{km/h}$?

        \[ v_{M}(0) = \answer[tolerance=0.01,onlineshowanswerbutton]{72} \, \mathrm{km/h} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x_M(t) = 20 \, \mathrm{m/s} \cdot t - 1{,}0 \, \mathrm{m/s^2} \cdot t^{2}$\\

            \underline{Gevraagd:} $v_M(0)$ in $\mathrm{km/h}$\\

            \underline{Oplossing:}\\
            De snelheid is de afgeleide van de positie:
            \[
                v_M(t) = x_M'(t) = 20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t
            \]
            Dus $v_M(0) = 20 \, \mathrm{m/s} = 20 \cdot 3{,}6 \, \mathrm{km/h} = 72 \, \mathrm{km/h}$.
        \end{solution}
    \end{question}

    \begin{question}
        Op welke tijdstippen $t_1$ en $t_2$ (met $t_1 < t_2$) bevinden de motor en de vrachtwagen zich op dezelfde plaats?

        \[ t_1 = \answer[tolerance=0.01,onlineshowanswerbutton]{3.0} \, \mathrm{s} \qquad \text{en} \qquad t_2 = \answer[tolerance=0.01,onlineshowanswerbutton]{9.0} \, \mathrm{s} \]

        \begin{hint}
            Wat geldt er voor de posities van beide voertuigen op het moment van een ontmoeting? Dit levert een vergelijking in $t$ op die je moet oplossen.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x_M(t) = 20 \, \mathrm{m/s} \cdot t - 1{,}0 \, \mathrm{m/s^2} \cdot t^{2}$ en $x_V(t) = 27 \, \mathrm{m} + 8{,}0 \, \mathrm{m/s} \cdot t$\\

            \underline{Gevraagd:} $t$ waarvoor $x_M = x_V$\\

            \underline{Oplossing:}\\
            We stellen de posities gelijk (eenheden even weggelaten, $t$ in $\mathrm{s}$):
            \[
                20t - t^{2} = 27 + 8t \quad \Rightarrow \quad t^{2} - 12t + 27 = 0
            \]
            \[
                t = \frac{12 \pm \sqrt{144 - 108}}{2} = \frac{12 \pm 6}{2} \quad \Rightarrow \quad t_1 = 3{,}0 \, \mathrm{s} \text{ en } t_2 = 9{,}0 \, \mathrm{s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken het snelheidsverschil $v_M - v_V$ op het eerste ontmoetingsmoment $t_1$.

        \[ v_M(t_1) - v_V(t_1) = \answer[tolerance=0.01,onlineshowanswerbutton]{6.0} \, \mathrm{m/s} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_M(t) = 20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_V(t) = 8{,}0 \, \mathrm{m/s}$; $t_1 = 3{,}0 \, \mathrm{s}$\\

            \underline{Gevraagd:} $v_M(t_1) - v_V(t_1)$\\

            \underline{Oplossing:}\\
            \begin{align*}
                v_M(3{,}0 \, \mathrm{s}) &= 20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot 3{,}0 \, \mathrm{s} = 14 \, \mathrm{m/s}\\
                v_M - v_V &= 14 \, \mathrm{m/s} - 8{,}0 \, \mathrm{m/s} = 6{,}0 \, \mathrm{m/s}
            \end{align*}
            De motor is op dat moment sneller dan de vrachtwagen en haalt hem in.
        \end{solution}
    \end{question}

    \begin{question}
        Wat gebeurt er op het tweede ontmoetingsmoment $t_2$?

        \wordChoice{
            \choice{De motor haalt de vrachtwagen in.}
            \choice[correct]{De vrachtwagen haalt de motor in.}
            \choice{De twee voertuigen kruisen elkaar in tegengestelde zin.}
        }

        \begin{hint}
            Vergelijk de snelheden van beide voertuigen op $t_2$. Wie is dan het snelst?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_M(t) = 20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_V(t) = 8{,}0 \, \mathrm{m/s}$; $t_2 = 9{,}0 \, \mathrm{s}$\\

            \underline{Gevraagd:} welk voertuig het andere inhaalt\\

            \underline{Oplossing:}\\
            \[
                v_M(9{,}0 \, \mathrm{s}) = 20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot 9{,}0 \, \mathrm{s} = 2{,}0 \, \mathrm{m/s}
            \]
            Dit is kleiner dan $v_V = 8{,}0 \, \mathrm{m/s}$. De vrachtwagen is dus sneller dan de motor en haalt hem in. Beide voertuigen rijden in dezelfde zin, dus ze kruisen elkaar niet in tegengestelde zin.
        \end{solution}
    \end{question}

    \begin{question}
        Wat is de grootste afstand tussen de motor en de vrachtwagen tijdens het tijdsinterval tussen $t_1$ en $t_2$?

        \[ \answer[tolerance=0.01,onlineshowanswerbutton]{9.0} \, \mathrm{m} \]

        \begin{hint}
            De afstand tussen beide voertuigen is eerst groeiend en daarna weer afnemend. Wat weet je over hun snelheden op het moment dat de afstand het grootst is?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_M(t) = 20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_V(t) = 8{,}0 \, \mathrm{m/s}$\\

            \underline{Gevraagd:} maximale afstand $x_M - x_V$\\

            \underline{Oplossing:}\\
            De afstand is maximaal als hij niet meer groeit en nog niet afneemt: dan zijn beide snelheden gelijk.
            \[
                20 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t = 8{,}0 \, \mathrm{m/s} \quad \Rightarrow \quad t = 6{,}0 \, \mathrm{s}
            \]
            De posities op dat moment zijn:
            \begin{align*}
                x_M(6{,}0 \, \mathrm{s}) &= 20 \, \mathrm{m/s} \cdot 6{,}0 \, \mathrm{s} - 1{,}0 \, \mathrm{m/s^2} \cdot (6{,}0 \, \mathrm{s})^{2} = 84 \, \mathrm{m}\\
                x_V(6{,}0 \, \mathrm{s}) &= 27 \, \mathrm{m} + 8{,}0 \, \mathrm{m/s} \cdot 6{,}0 \, \mathrm{s} = 75 \, \mathrm{m}
            \end{align*}
            De grootste afstand is $84 \, \mathrm{m} - 75 \, \mathrm{m} = 9{,}0 \, \mathrm{m}$.
        \end{solution}
    \end{question}
\end{exercise}

\section{Kromlijnige beweging: componenten combineren}
De $x$- en $y$-componenten van snelheid en versnelling combineer je met de stelling van Pythagoras.

%———————————————————————————————————————
% OEFENING: kromlijnige beweging met functievoorschriften, Pythagoras
\begin{exercise}
    Een drone vertrekt uit de oorsprong. Zijn positie ten opzichte van een horizontale $x$-as en een verticale $y$-as wordt gegeven door
    \begin{align*}
        x(t) &= 4{,}0 \, \mathrm{\frac{m}{s}} \cdot t\\
        y(t) &= 1{,}5 \, \mathrm{\frac{m}{s^2}} \cdot t^{2}
    \end{align*}
    met $t$ in seconden.

    \begin{question}
        Bereken de grootte van de snelheid van de drone op $t = 2{,}0 \, \mathrm{s}$.

        \[ v = \answer[tolerance=0.01,onlineshowanswerbutton]{7.2} \, \mathrm{m/s} \]

        \begin{hint}
            Bepaal eerst de componenten $v_x$ en $v_y$. Hoe combineer je die tot de grootte van de snelheid?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = 4{,}0 \, \mathrm{m/s} \cdot t$ en $y(t) = 1{,}5 \, \mathrm{m/s^2} \cdot t^{2}$; $t = 2{,}0 \, \mathrm{s}$\\

            \underline{Gevraagd:} $v$\\

            \underline{Oplossing:}\\
            \begin{align*}
                v_x(t) &= x'(t) = 4{,}0 \, \mathrm{m/s}\\
                v_y(t) &= y'(t) = 3{,}0 \, \mathrm{m/s^2} \cdot t \quad \Rightarrow \quad v_y(2{,}0 \, \mathrm{s}) = 6{,}0 \, \mathrm{m/s}
            \end{align*}
            De componenten staan loodrecht op elkaar, dus we gebruiken de stelling van Pythagoras:
            \[
                v = \sqrt{v_x^{2} + v_y^{2}} = \sqrt{(4{,}0 \, \mathrm{m/s})^{2} + (6{,}0 \, \mathrm{m/s})^{2}} = 7{,}2 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de grootte van de versnelling van de drone. Hangt die af van het tijdstip?

        \[ a = \answer[tolerance=0.01,onlineshowanswerbutton]{3.0} \, \mathrm{m/s^2} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 4{,}0 \, \mathrm{m/s}$ en $v_y(t) = 3{,}0 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $a$\\

            \underline{Oplossing:}\\
            \begin{align*}
                a_x(t) &= v_x'(t) = 0 \, \mathrm{m/s^2}\\
                a_y(t) &= v_y'(t) = 3{,}0 \, \mathrm{m/s^2}
            \end{align*}
            \[
                a = \sqrt{a_x^{2} + a_y^{2}} = \sqrt{(0 \, \mathrm{m/s^2})^{2} + (3{,}0 \, \mathrm{m/s^2})^{2}} = 3{,}0 \, \mathrm{m/s^2}
            \]
            De versnelling is constant en wijst steeds in de positieve $y$-zin.
        \end{solution}
    \end{question}

    \begin{question}
        Op welk tijdstip bedraagt de grootte van de snelheid $10 \, \mathrm{m/s}$?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{3.1} \, \mathrm{s} \]

        \begin{hint}
            Schrijf de grootte van de snelheid uit als functie van $t$ en stel ze gelijk aan $10 \, \mathrm{m/s}$.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x = 4{,}0 \, \mathrm{m/s}$ en $v_y(t) = 3{,}0 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $t$ waarvoor $v = 10 \, \mathrm{m/s}$\\

            \underline{Oplossing:}\\
            We stellen $\sqrt{v_x^{2} + v_y^{2}} = 10$ (eenheden even weggelaten, $t$ in $\mathrm{s}$) en kwadrateren:
            \[
                16 + 9t^{2} = 100 \quad \Rightarrow \quad t^{2} = \frac{84}{9} \quad \Rightarrow \quad t = 3{,}1 \, \mathrm{s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Op welke hoogte $y$ bevindt de drone zich op het moment dat hij $12 \, \mathrm{m}$ horizontaal heeft afgelegd?

        \[ y = \answer[tolerance=0.01,onlineshowanswerbutton]{13.5} \, \mathrm{m} \]

        \begin{hint}
            Op welk tijdstip is $x = 12 \, \mathrm{m}$?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x = 12 \, \mathrm{m}$\\

            \underline{Gevraagd:} $y$\\

            \underline{Oplossing:}\\
            \[
                12 \, \mathrm{m} = 4{,}0 \, \mathrm{m/s} \cdot t \quad \Rightarrow \quad t = 3{,}0 \, \mathrm{s}
            \]
            \[
                y(3{,}0 \, \mathrm{s}) = 1{,}5 \, \mathrm{m/s^2} \cdot (3{,}0 \, \mathrm{s})^{2} = 13{,}5 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING 7 (moeilijkst): kromlijnige beweging, tangentiele en normale versnelling
\begin{exercise}
    Een bal wordt weggeschopt. Zijn positie wordt beschreven ten opzichte van een horizontale $x$-as (in de bewegingsrichting) en een verticale $y$-as (omhoog) door
    \begin{align*}
        x(t) &= 5{,}0 \, \mathrm{\frac{m}{s}} \cdot t\\
        y(t) &= 2{,}0 \, \mathrm{m} + 12 \, \mathrm{\frac{m}{s}} \cdot t - 5{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t^{2}
    \end{align*}
    met $t$ in seconden. Deze vergelijkingen gelden tot de bal de grond ($y = 0{,}0 \, \mathrm{m}$) raakt.

    \begin{question}
        Bereken de grootte van de snelheid van de bal op $t = 0{,}50 \, \mathrm{s}$.

        \[ v = \answer[tolerance=0.01,onlineshowanswerbutton]{8.6} \, \mathrm{m/s} \]

        \begin{hint}
            De snelheidsvector heeft een $x$- en een $y$-component. Hoe vind je elke component vanuit de positievergelijking?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = 5{,}0 \, \mathrm{m/s} \cdot t$ en $y(t) = 2{,}0 \, \mathrm{m} + 12 \, \mathrm{m/s} \cdot t - 5{,}0 \, \mathrm{m/s^2} \cdot t^{2}$; $t = 0{,}50 \, \mathrm{s}$\\

            \underline{Gevraagd:} $v$\\

            \underline{Oplossing:}\\
            De componenten van de snelheid zijn de afgeleiden van de posities:
            \begin{align*}
                v_x(t) &= x'(t) = 5{,}0 \, \mathrm{m/s}\\
                v_y(t) &= y'(t) = 12 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot t
            \end{align*}
            Op $t = 0{,}50 \, \mathrm{s}$ is $v_y = 12 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot 0{,}50 \, \mathrm{s} = 7{,}0 \, \mathrm{m/s}$. De grootte van de snelheid is:
            \[
                v = \sqrt{v_x^{2} + v_y^{2}} = \sqrt{(5{,}0 \, \mathrm{m/s})^{2} + (7{,}0 \, \mathrm{m/s})^{2}} = 8{,}6 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Ontbind de versnelling van de bal op $t = 0{,}50 \, \mathrm{s}$ in een tangentiële component $a_t$ (evenwijdig met de snelheid) en een normale component $a_n$ (loodrecht op de snelheid). Een negatieve $a_t$ betekent dat de component tegengesteld is aan de snelheid.

        \[ a_t = \answer[tolerance=0.02,onlineshowanswerbutton]{-8.1} \, \mathrm{m/s^2} \qquad \text{en} \qquad a_n = \answer[tolerance=0.02,onlineshowanswerbutton]{5.8} \, \mathrm{m/s^2} \]

        \begin{hint}
            Bepaal eerst de versnellingsvector zelf, en teken daarna de snelheids- en versnellingsvector samen in één schets.
        \end{hint}
        \begin{hint}
            Welke hoek maken de snelheidsvector en de versnellingsvector met elkaar? De component van $\vec{\mathbf{a}}$ langs en loodrecht op $\vec{\mathbf{v}}$ volgt uit die hoek.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x = 5{,}0 \, \mathrm{m/s}$ en $v_y = 7{,}0 \, \mathrm{m/s}$ op $t = 0{,}50 \, \mathrm{s}$\\

            \underline{Gevraagd:} $a_t$ en $a_n$\\

            \underline{Oplossing:}\\
            De componenten van de versnelling zijn de afgeleiden van de snelheidscomponenten:
            \begin{align*}
                a_x(t) &= v_x'(t) = 0 \, \mathrm{m/s^2}\\
                a_y(t) &= v_y'(t) = -10 \, \mathrm{m/s^2}
            \end{align*}
            De versnelling wijst dus recht naar beneden, met grootte $a = 10 \, \mathrm{m/s^2}$. De snelheidsvector maakt een hoek met de $x$-as van:
            \[
                \theta = \tan^{-1}\left(\frac{7{,}0 \, \mathrm{m/s}}{5{,}0 \, \mathrm{m/s}}\right) = 54{,}5^\circ
            \]
            \begin{center}
                \begin{tikzpicture}[scale=1]
                    \coordinate (P) at (0,0);
                    \coordinate (O) at (-3.2,-2.6);
                    \tekenassen[1.2]{O}{0}
                    \draw[snelheid] (P) -- (54.46:2.8) node[above right] {$\vec{\mathbf{v}}$};
                    \draw[versnelling] (P) -- (0,-2.4) node[below] {$\vec{\mathbf{a}}$};
                    \draw[versnellingcomponent] (P) -- (234.46:1.95) node[below left] {$\vec{\mathbf{a}}_{t}$};
                    \draw[versnellingcomponent] (P) -- (-35.54:1.4) node[right] {$\vec{\mathbf{a}}_{n}$};
                    \draw[gray, dashed, thin] (234.46:1.95) -- (0,-2.4) -- (-35.54:1.4);
                    \fill (P) circle (1.5pt);
                \end{tikzpicture}

                Schets (niet op schaal)
            \end{center}
            De versnelling wijst naar beneden ($-90^\circ$), dus de hoek tussen $\vec{\mathbf{v}}$ en $\vec{\mathbf{a}}$ is $\varphi = 90^\circ + 54{,}5^\circ = 144{,}5^\circ$. Daaruit volgt:
            \begin{align*}
                a_t &= a \cdot \cos(\varphi) = 10 \, \mathrm{m/s^2} \cdot \cos(144{,}5^\circ) = -8{,}1 \, \mathrm{m/s^2}\\
                a_n &= a \cdot \sin(\varphi) = 10 \, \mathrm{m/s^2} \cdot \sin(144{,}5^\circ) = 5{,}8 \, \mathrm{m/s^2}
            \end{align*}
            Controle: $\sqrt{(8{,}1 \, \mathrm{m/s^2})^{2} + (5{,}8 \, \mathrm{m/s^2})^{2}} \simeq 10 \, \mathrm{m/s^2}$. De tangentiële component is negatief: de bal vertraagt op dat moment.
        \end{solution}
    \end{question}

    \begin{question}
        Op welk tijdstip $t$ is de snelheid van de bal het kleinst, en hoe groot is die snelheid dan?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{1.2} \, \mathrm{s} \qquad \text{en} \qquad v_{\min} = \answer[tolerance=0.01,onlineshowanswerbutton]{5.0} \, \mathrm{m/s} \]

        \begin{hint}
            De grootte van de snelheid hangt af van $v_x$ en $v_y$. Welke van beide componenten verandert er in de tijd, en wat is dus de kleinst mogelijke waarde ervan?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x = 5{,}0 \, \mathrm{m/s}$ (constant) en $v_y(t) = 12 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $t$ en $v_{\min}$\\

            \underline{Oplossing:}\\
            De snelheid is $v = \sqrt{v_x^{2} + v_y^{2}}$. Omdat $v_x$ constant is, is $v$ minimaal als $\lvert v_y \rvert$ minimaal is, dus als $v_y = 0$:
            \[
                12 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot t = 0 \quad \Rightarrow \quad t = 1{,}2 \, \mathrm{s}
            \]
            De snelheid is dan enkel nog horizontaal: $v_{\min} = v_x = 5{,}0 \, \mathrm{m/s}$.

            Op dat moment staat $\vec{\mathbf{a}}$ (verticaal) loodrecht op $\vec{\mathbf{v}}$ (horizontaal). Er geldt dus $a_t = 0$ en $a_n = 10 \, \mathrm{m/s^2}$: de snelheid verandert enkel van richting, niet van grootte.
        \end{solution}
    \end{question}

    \begin{question}
        Op welke horizontale positie $x$ raakt de bal de grond?

        \[ x = \answer[tolerance=0.01,onlineshowanswerbutton]{12.8} \, \mathrm{m} \]

        \begin{hint}
            Welke waarde heeft $y$ op het moment dat de bal de grond raakt? Los eerst op naar de tijd.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $x(t) = 5{,}0 \, \mathrm{m/s} \cdot t$ en $y(t) = 2{,}0 \, \mathrm{m} + 12 \, \mathrm{m/s} \cdot t - 5{,}0 \, \mathrm{m/s^2} \cdot t^{2}$\\

            \underline{Gevraagd:} $x$ als $y = 0$\\

            \underline{Oplossing:}\\
            We stellen $y(t) = 0$ (eenheden even weggelaten, $t$ in $\mathrm{s}$):
            \[
                5t^{2} - 12t - 2 = 0 \quad \Rightarrow \quad t = \frac{12 \pm \sqrt{144 + 40}}{10} = \frac{12 \pm 13{,}56}{10}
            \]
            Enkel de positieve oplossing heeft fysische betekenis: $t = 2{,}56 \, \mathrm{s}$. Dan is:
            \[
                x = 5{,}0 \, \mathrm{m/s} \cdot 2{,}56 \, \mathrm{s} = 12{,}8 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING: meetgegevens 100 m
\begin{exercise}
    Tijdens een $100 \, \mathrm{m}$-sprint werden op enkele tijdstippen de positie $x$ en de snelheid $v_x$ van de loper gemeten.

    \begin{center}
        \begin{tabular}{|c|c|c|}
            \hline
            $t\,(\mathrm{s})$ & $x\,(\mathrm{m})$ & $v_x\,(\mathrm{m/s})$ \\
            \hline
            $0{,}00$ & $0{,}00$ & $0{,}00$ \\
            $4{,}62$ & $39{,}88$ & $11{,}63$ \\
            $5{,}79$ & $53{,}94$ & $12{,}11$ \\
            $7{,}10$ & $69{,}88$ & $12{,}35$ \\
            $8{,}19$ & $83{,}29$ & $12{,}08$ \\
            $9{,}58$ & $100{,}00$ & $11{,}80$ \\
            \hline
        \end{tabular}
    \end{center}

    \begin{question}
        Bereken de gemiddelde snelheid $\langle v_x \rangle$ tussen $t = 5{,}79 \, \mathrm{s}$ en $t = 8{,}19 \, \mathrm{s}$.

        \[ \langle v_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{12.2} \, \mathrm{m/s} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & x(5{,}79 \, \mathrm{s}) = 53{,}94 \, \mathrm{m} & & x(8{,}19 \, \mathrm{s}) = 83{,}29 \, \mathrm{m}
            \end{align*}

            \underline{Gevraagd:} $\langle v_x \rangle$\\

            \underline{Oplossing:}\\
            \[
                \langle v_x \rangle = \frac{\Delta x}{\Delta t} = \frac{83{,}29 \, \mathrm{m} - 53{,}94 \, \mathrm{m}}{8{,}19 \, \mathrm{s} - 5{,}79 \, \mathrm{s}} = 12{,}2 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde versnelling $\langle a_x \rangle$ tussen $t = 7{,}10 \, \mathrm{s}$ en $t = 9{,}58 \, \mathrm{s}$. Wat zegt het teken over de loper?

        \[ \langle a_x \rangle = \answer[tolerance=0.02,onlineshowanswerbutton]{-0.22} \, \mathrm{m/s^2} \]

        De loper: \wordChoice{
            \choice{versnelt nog steeds}
            \choice[correct]{vertraagt licht}
        }

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & v_x(7{,}10 \, \mathrm{s}) = 12{,}35 \, \mathrm{m/s} & & v_x(9{,}58 \, \mathrm{s}) = 11{,}80 \, \mathrm{m/s}
            \end{align*}

            \underline{Gevraagd:} $\langle a_x \rangle$\\

            \underline{Oplossing:}\\
            \[
                \langle a_x \rangle = \frac{\Delta v_x}{\Delta t} = \frac{11{,}80 \, \mathrm{m/s} - 12{,}35 \, \mathrm{m/s}}{9{,}58 \, \mathrm{s} - 7{,}10 \, \mathrm{s}} = -0{,}22 \, \mathrm{m/s^2}
            \]
            De snelheid is positief en de versnelling negatief: de loper vertraagt in dit interval licht.
        \end{solution}
    \end{question}

    \begin{question}
        Bereken de gemiddelde snelheid over de volledige race in $\mathrm{m/s}$ en in $\mathrm{km/h}$. Waarom ligt die lager dan elke snelheid uit de tabel vanaf $t = 4{,}62 \, \mathrm{s}$?

        \[ \langle v_x \rangle = \answer[tolerance=0.01,onlineshowanswerbutton]{10.4} \, \mathrm{m/s} = \answer[tolerance=0.01,onlineshowanswerbutton]{37.6} \, \mathrm{km/h} \]

        \begin{solution}\nl
            \underline{Gegevens:}
            \begin{align*}
                & \Delta x = 100{,}00 \, \mathrm{m} & & \Delta t = 9{,}58 \, \mathrm{s}
            \end{align*}

            \underline{Gevraagd:} $\langle v_x \rangle$\\

            \underline{Oplossing:}\\
            \begin{align*}
                \langle v_x \rangle &= \frac{100{,}00 \, \mathrm{m}}{9{,}58 \, \mathrm{s}} = 10{,}4 \, \mathrm{m/s}\\
                &= 10{,}4 \cdot 3{,}6 \, \mathrm{km/h} = 37{,}6 \, \mathrm{km/h}
            \end{align*}
            De loper start uit rust en moet eerst op snelheid komen. In de beginfase is de snelheid veel kleiner dan $12 \, \mathrm{m/s}$, en dat drukt de gemiddelde snelheid over de hele race.
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING: verticale worp met positievergelijking
\begin{exercise}
    Een steen wordt vanaf de grond verticaal omhoog geworpen. Zijn hoogte $y$ boven de grond wordt (tot hij terug de grond raakt) gegeven door
    \[
        y(t) = 20 \, \mathrm{\frac{m}{s}} \cdot t - 5{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t^{2}
    \]
    met $t$ in seconden. De $y$-as wijst naar boven.

    \begin{question}
        Op welk tijdstip bereikt de steen het hoogste punt, en hoe hoog is dat?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{2.0} \, \mathrm{s} \qquad \text{en} \qquad y_{\max} = \answer[tolerance=0.01,onlineshowanswerbutton]{20} \, \mathrm{m} \]

        \begin{hint}
            Wat is de snelheid van de steen in het hoogste punt?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $y(t) = 20 \, \mathrm{m/s} \cdot t - 5{,}0 \, \mathrm{m/s^2} \cdot t^{2}$\\

            \underline{Gevraagd:} $t$ en $y_{\max}$\\

            \underline{Oplossing:}\\
            De snelheid is de afgeleide van de positie:
            \[
                v_y(t) = y'(t) = 20 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot t
            \]
            In het hoogste punt keert de steen van zin om, dus $v_y = 0$:
            \[
                20 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot t = 0 \quad \Rightarrow \quad t = 2{,}0 \, \mathrm{s}
            \]
            \[
                y(2{,}0 \, \mathrm{s}) = 20 \, \mathrm{m/s} \cdot 2{,}0 \, \mathrm{s} - 5{,}0 \, \mathrm{m/s^2} \cdot (2{,}0 \, \mathrm{s})^{2} = 20 \, \mathrm{m}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Op welke twee tijdstippen $t_1$ en $t_2$ (met $t_1 < t_2$) bevindt de steen zich op $15 \, \mathrm{m}$ hoogte?

        \[ t_1 = \answer[tolerance=0.01,onlineshowanswerbutton]{1.0} \, \mathrm{s} \qquad \text{en} \qquad t_2 = \answer[tolerance=0.01,onlineshowanswerbutton]{3.0} \, \mathrm{s} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $y = 15 \, \mathrm{m}$\\

            \underline{Gevraagd:} $t_1$ en $t_2$\\

            \underline{Oplossing:}\\
            We stellen $y(t) = 15$ (eenheden even weggelaten, $t$ in $\mathrm{s}$):
            \[
                20t - 5t^{2} = 15 \quad \Rightarrow \quad t^{2} - 4t + 3 = 0 \quad \Rightarrow \quad t = \frac{4 \pm \sqrt{16 - 12}}{2}
            \]
            Dus $t_1 = 1{,}0 \, \mathrm{s}$ (op weg naar boven) en $t_2 = 3{,}0 \, \mathrm{s}$ (op weg naar beneden).
        \end{solution}
    \end{question}

    \begin{question}
        Met welke snelheid $v_y$ komt de steen terug op de grond, en hoe groot is de afgelegde weg $s$ tot dan?

        \[ v_y = \answer[tolerance=0.01,onlineshowanswerbutton]{-20} \, \mathrm{m/s} \qquad \text{en} \qquad s = \answer[tolerance=0.01,onlineshowanswerbutton]{40} \, \mathrm{m} \]

        \begin{hint}
            Wanneer is $y$ opnieuw nul? Is de verplaatsing dan gelijk aan de afgelegde weg?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $y(t) = 20 \, \mathrm{m/s} \cdot t - 5{,}0 \, \mathrm{m/s^2} \cdot t^{2}$ en $v_y(t) = 20 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $v_y$ bij de landing en $s$\\

            \underline{Oplossing:}\\
            We stellen $y = 0$ (eenheden even weggelaten): $20t - 5t^{2} = 5t \cdot (4 - t) = 0$, dus de steen landt op $t = 4{,}0 \, \mathrm{s}$.
            \[
                v_y(4{,}0 \, \mathrm{s}) = 20 \, \mathrm{m/s} - 10 \, \mathrm{m/s^2} \cdot 4{,}0 \, \mathrm{s} = -20 \, \mathrm{m/s}
            \]
            De steen gaat $20 \, \mathrm{m}$ omhoog en daarna $20 \, \mathrm{m}$ omlaag:
            \[
                s = 20 \, \mathrm{m} + 20 \, \mathrm{m} = 40 \, \mathrm{m}
            \]
            De verplaatsing is nul, de afgelegde weg niet.
        \end{solution}
    \end{question}
\end{exercise}

% OEFENING (moeilijkst van de extra's): kleinste snelheid via afgeleide van v^2
\begin{exercise}
    De snelheidscomponenten van een voorwerp in het $(x,y)$-vlak worden gegeven door
    \begin{align*}
        v_x(t) &= 3{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t\\
        v_y(t) &= 4{,}0 \, \mathrm{\frac{m}{s}} - 2{,}0 \, \mathrm{\frac{m}{s^2}} \cdot t
    \end{align*}
    met $t$ in seconden.

    \begin{question}
        Bereken de grootte van de versnelling van het voorwerp.

        \[ a = \answer[tolerance=0.01,onlineshowanswerbutton]{3.6} \, \mathrm{m/s^2} \]

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 3{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_y(t) = 4{,}0 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $a$\\

            \underline{Oplossing:}\\
            \begin{align*}
                a_x &= v_x'(t) = 3{,}0 \, \mathrm{m/s^2}\\
                a_y &= v_y'(t) = -2{,}0 \, \mathrm{m/s^2}
            \end{align*}
            \[
                a = \sqrt{a_x^{2} + a_y^{2}} = \sqrt{(3{,}0 \, \mathrm{m/s^2})^{2} + (-2{,}0 \, \mathrm{m/s^2})^{2}} = 3{,}6 \, \mathrm{m/s^2}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Op welk tijdstip is de grootte van de snelheid het kleinst, en hoe groot is ze dan?

        \[ t = \answer[tolerance=0.01,onlineshowanswerbutton]{0.62} \, \mathrm{s} \qquad \text{en} \qquad v_{\min} = \answer[tolerance=0.01,onlineshowanswerbutton]{3.3} \, \mathrm{m/s} \]

        \begin{hint}
            Nu veranderen zowel $v_x$ als $v_y$ in de tijd, dus de kleinste snelheid is niet meer het moment waarop een component nul is. Schrijf $v^2$ uit als functie van $t$: dat is een eenvoudiger functie om te minimaliseren.
        \end{hint}
        \begin{hint}
            Hoe zoek je met afgeleiden naar het minimum van een functie?
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x(t) = 3{,}0 \, \mathrm{m/s^2} \cdot t$ en $v_y(t) = 4{,}0 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot t$\\

            \underline{Gevraagd:} $t$ en $v_{\min}$\\

            \underline{Oplossing:}\\
            Met de stelling van Pythagoras is $v^{2} = v_x^{2} + v_y^{2}$. We laten de eenheden even weg ($t$ in $\mathrm{s}$):
            \[
                v^{2} = 9t^{2} + (4 - 2t)^{2} = 13t^{2} - 16t + 16
            \]
            De snelheid is het kleinst als $v^{2}$ het kleinst is, dus als de afgeleide nul is:
            \[
                (v^{2})' = 26t - 16 = 0 \quad \Rightarrow \quad t = \frac{16}{26} = 0{,}62 \, \mathrm{s}
            \]
            Invullen:
            \begin{align*}
                v_x &= 3{,}0 \, \mathrm{m/s^2} \cdot 0{,}615 \, \mathrm{s} = 1{,}85 \, \mathrm{m/s}\\
                v_y &= 4{,}0 \, \mathrm{m/s} - 2{,}0 \, \mathrm{m/s^2} \cdot 0{,}615 \, \mathrm{s} = 2{,}77 \, \mathrm{m/s}
            \end{align*}
            \[
                v_{\min} = \sqrt{(1{,}85 \, \mathrm{m/s})^{2} + (2{,}77 \, \mathrm{m/s})^{2}} = 3{,}3 \, \mathrm{m/s}
            \]
        \end{solution}
    \end{question}

    \begin{question}
        Op het moment van de kleinste snelheid geldt: \wordChoice{
            \choice[correct]{de versnelling staat loodrecht op de snelheid}
            \choice{de versnelling is evenwijdig met de snelheid}
            \choice{de versnelling is nul}
        }

        \begin{hint}
            Bepaal de hoek die $\vec{\mathbf{v}}$ en $\vec{\mathbf{a}}$ elk maken met de $x$-as.
        \end{hint}

        \begin{solution}\nl
            \underline{Gegevens:}\\
            $v_x = 1{,}85 \, \mathrm{m/s}$, $v_y = 2{,}77 \, \mathrm{m/s}$, $a_x = 3{,}0 \, \mathrm{m/s^2}$ en $a_y = -2{,}0 \, \mathrm{m/s^2}$\\

            \underline{Gevraagd:} de onderlinge stand van $\vec{\mathbf{v}}$ en $\vec{\mathbf{a}}$\\

            \underline{Oplossing:}\\
            \begin{align*}
                \theta_v &= \tan^{-1}\left(\frac{2{,}77 \, \mathrm{m/s}}{1{,}85 \, \mathrm{m/s}}\right) = 56{,}3^\circ\\
                \theta_a &= \tan^{-1}\left(\frac{-2{,}0 \, \mathrm{m/s^2}}{3{,}0 \, \mathrm{m/s^2}}\right) = -33{,}7^\circ
            \end{align*}
            Het verschil is $56{,}3^\circ - (-33{,}7^\circ) = 90^\circ$. De versnelling staat dus loodrecht op de snelheid: de tangentiële versnelling is nul ($a_t = 0$) en de snelheid verandert op dat moment enkel van richting, niet van grootte. Dat is precies waarom de grootte van de snelheid daar minimaal is.
        \end{solution}
    \end{question}
\end{exercise}
